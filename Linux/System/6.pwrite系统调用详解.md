# pwrite 系统调用详解

## 1. 问题背景

pwrite（pread）是 POSIX 标准的文件写入/读取接口，支持原子性地在指定偏移位置执行 IO 操作，是高性能 IO 的基础系统调用。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  pwrite/pread 核心价值：                                   │
│                                                             │
│  • 原子性：不受当前文件偏移影响                           │
│  • 并发安全：多线程可同时操作不同位置                     │
│  • 高性能：减少 lseek + read/write 调用                  │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

---

## 2. 问题现象

| 现象 | 描述 | 影响 |
|------|------|------|
| 文件偏移竞争 | lseek + write 不是原子操作 | 数据覆盖风险 |
| 多线程不安全 | 共享文件描述符时偏移混乱 | 并发写入出错 |
| IO 调用多 | 每次 IO 需额外系统调用 | 性能下降 |

## 4. 原因分析

### 4.1 write vs pwrite 对比

```c
// 传统 write（依赖文件偏移）
ssize_t write(int fd, const void *buf, size_t count) {
    // 1. 获取当前文件偏移
    off_t current = lseek(fd, 0, SEEK_CUR);
    
    // 2. 写入数据
    return actual_write(fd, buf, count, current);
}

// pwrite（指定偏移，不改变文件偏移）
ssize_t pwrite(int fd, const void *buf, size_t count, off_t offset) {
    // 直接在指定偏移写入
    return actual_write(fd, buf, count, offset);
}
```

### 4.2 非原子性风险

```c
// 多线程使用 write 的问题
// Thread 1:
// lseek(fd, 1000, SEEK_SET);  // 偏移到 1000
// write(fd, buf1, 100);        // 写入 100 字节

// Thread 2:
// lseek(fd, 2000, SEEK_SET);  // 偏移到 2000
// write(fd, buf2, 100);       // 写入 100 字节

// 实际情况可能：
// Thread 1: lseek(fd, 1000) → 偏移=1000
// Thread 2: lseek(fd, 2000) → 偏移=2000
// Thread 1: write(fd, buf1, 100) → 写入位置=2000（被覆盖！）
// Thread 2: write(fd, buf2, 100) → 写入位置=2000

// 使用 pwrite 避免此问题
pwrite(fd, buf1, 100, 1000);  // 原子地在 1000 位置写
pwrite(fd, buf2, 100, 2000);  // 原子地在 2000 位置写
```

## 5. 解决方案

### 5.1 pwrite/pread 函数原型

```c
#include <sys/types.h>
#include <unistd.h>

// 写入
ssize_t pwrite(int fd, const void *buf, size_t count, off_t offset);

// 读取
ssize_t pread(int fd, void *buf, size_t count, off_t offset);

// 返回值：
// 成功：实际写入/读取的字节数
// 失败：-1，errno 设置
```

### 5.2 内核实现流程


### 5.3 代码实现

```c
// pwrite 示例代码

#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
#include <string.h>

int main() {
    const char *filename = "test.txt";
    const char *data = "Hello, pwrite!";
    
    // 打开文件
    int fd = open(filename, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd < 0) {
        perror("open");
        return 1;
    }
    
    // 使用 pwrite 在指定偏移写入
    off_t offset = 0;
    ssize_t written = pwrite(fd, data, strlen(data), offset);
    
    if (written < 0) {
        perror("pwrite");
        close(fd);
        return 1;
    }
    
    printf("Written %zd bytes at offset %ld\n", written, offset);
    
    // 追加写入（文件末尾）
    offset = lseek(fd, 0, SEEK_END);  // 获取文件大小作为偏移
    written = pwrite(fd, "\nAppended text", 14, offset);
    
    printf("Written %zd bytes at end\n", written);
    
    close(fd);
    return 0;
}
```

## 7. 推荐方案

### 7.1 高性能场景

```c
// 推荐使用 pwrite + O_DIRECT（绕过 Page Cache）

int fd = open("file", O_WRONLY | O_DIRECT | O_CREAT, 0644);

// 注意：O_DIRECT 需要对齐的缓冲区
alignas(512) char buf[4096];
memcpy(buf, data, len);

pwrite(fd, buf, len, offset);
```

### 7.2 并发写入场景

```c
// 多线程并发写入不同位置

#include <pthread.h>

typedef struct {
    int fd;
    const char *data;
    size_t len;
    off_t offset;
} WriteArgs;

void* write_thread(void *arg) {
    WriteArgs *args = (WriteArgs *)arg;
    
    // pwrite 是线程安全的
    ssize_t written = pwrite(args->fd, args->data, args->len, args->offset);
    
    return (void*)(intptr_t)written;
}

int main() {
    pthread_t threads[4];
    WriteArgs args[4];
    
    // 4 个线程同时写入不同位置
    for (int i = 0; i < 4; i++) {
        args[i].fd = fd;
        args[i].data = "Data";
        args[i].len = 4;
        args[i].offset = i * 100;
        
        pthread_create(&threads[i], NULL, write_thread, &args[i]);
    }
    
    // 等待完成
    for (int i = 0; i < 4; i++) {
        pthread_join(threads[i], NULL);
    }
    
    return 0;
}
```

## 8. 实施步骤

### 8.1 代码迁移

```bash
# 1. 搜索使用 lseek + write 的代码
grep -rn "lseek.*write" --include="*.c" --include="*.cpp" .

# 2. 替换为 pwrite
# 原代码：
lseek(fd, offset, SEEK_SET);
write(fd, buf, count);

// 新代码：
pwrite(fd, buf, count, offset);
```

### 8.2 注意事项

```c
// pwrite 注意点

// 1. 偏移量不能为负
if (offset < 0) {
    errno = EINVAL;
    return -1;
}

// 2. 部分写入需处理
ssize_t written = pwrite(fd, buf, count, offset);
if (written > 0 && written < count) {
    // 部分写入，可能需要重试剩余部分
    ssize_t remaining = count - written;
    written += pwrite(fd, buf + written, remaining, offset + written);
}

// 3. 偏移超出文件末尾
// 自动扩展文件（与 lseek + write 行为一致）
```

## 9. 风险与注意事项

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
| 部分写入 | 数据不完整 | 检查返回值，重试 |
| 偏移越界 | EINVAL | 参数校验 |
| 并发文件锁 | 写入阻塞 | 合理使用锁 |
| O_DIRECT 对齐 | 需 512 字节对齐 | 使用 posix_memalign |

## 10. 验证方式

```c
// pwrite 正确性测试

#include <assert.h>
#include <stdio.h>

void test_pwrite() {
    int fd = open("test.bin", O_RDWR | O_CREAT | O_TRUNC, 0644);
    
    // 测试 1：基本写入
    char buf1[] = "ABCD";
    pwrite(fd, buf1, 4, 0);
    
    // 测试 2：覆盖写入
    char buf2[] = "XYZ";
    pwrite(fd, buf2, 3, 1);
    
    // 验证结果
    char result[10] = {0};
    pread(fd, result, 10, 0);
    
    // 应该是 "AXYZ"（A被X覆盖，后面追加YZ）
    // 等等，应该是 "AXYZ" + 未定义...
    // 正确理解：
    // 初始：写入 "ABCD" → [A][B][C][D]
    // 覆盖：写入 "XYZ" 到偏移1 → [A][X][Y][Z]
    
    printf("Result: %.4s\n", result);
    assert(result[0] == 'A');
    assert(result[1] == 'X');
    assert(result[2] == 'Y');
    assert(result[3] == 'Z');
    
    close(fd);
    printf("All tests passed!\n");
}
```

## 11. 总结

1. **pwrite 定义**：原子性地在指定偏移写入文件
2. **核心优势**：不受文件偏移影响，并发安全，高性能
3. **内核流程**：syscall → VFS → 文件系统 → 块设备
4. **使用场景**：数据库、日志、消息队列
5. **注意问题**：部分写入处理、O_DIRECT 对齐要求

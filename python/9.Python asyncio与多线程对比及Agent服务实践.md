# 9. Python asyncio 与多线程对比及 Agent 服务实践

## 1. 问题背景

Python 的 asyncio 和多线程是两种主要的并发模型。Agent 服务通常涉及大量 I/O 操作（HTTP 请求、数据库查询、区块链 RPC 调用），选择合适的并发模型直接影响服务性能和稳定性。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  并发模型对比：                                            │
│                                                             │
│  asyncio（协程）：                                         │
│  • 单线程 + 事件循环                                      │
│  • 协作式多任务                                          │
│  • 适合 I/O 密集型                                       │
│  • 无 GIL 问题                                            │
│                                                             │
│  多线程：                                                  │
│  • 多线程 + 抢占式调度                                    │
│  • 竞争式多任务                                          │
│  • CPU/IO 混合型                                         │
│  • GIL 限制                                               │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

---

## 2. 问题现象

| 现象 | asyncio | 多线程 |
|------|---------|--------|
| 响应延迟 | 低（单请求快） | 中（线程切换开销） |
| 吞吐量 | 高（I/O密集） | 低（GIL限制） |
| CPU 利用率 | 单核 | 多核 |
| 复杂度 | 异步陷阱多 | 竞态条件多 |

---

## 3. 影响范围

- HTTP API 服务（FastAPI）
- 数据库连接池
- 区块链 RPC 调用
- Agent 任务调度

## 4. 原因分析

### 4.1 GIL 限制分析

```python
"""
GIL (Global Interpreter Lock) 的影响：

Python 多线程的问题：
• 同一时刻只有一个线程执行 Python 字节码
• CPU 密集型任务无法并行
• I/O 密集型任务受限于 GIL 释放时机
"""

import threading
import time

def cpu_task(n):
    """CPU 密集型任务"""
    result = 0
    for i in range(n):
        result += i * i
    return result

# 多线程 vs 单线程（CPU 密集型）
def test_gil():
    start = time.time()
    
    # 2 个线程
    threads = [threading.Thread(target=cpu_task, args=(10_000_000,)) for _ in range(2)]
    for t in threads: t.start()
    for t in threads: t.join()
    
    print(f"2 threads: {time.time() - start:.2f}s")
    # 结果：约 2x 单线程时间，说明 GIL 限制了并行
    
    start = time.time()
    cpu_task(20_000_000)  # 单线程执行 2 倍工作量
    print(f"1 thread: {time.time() - start:.2f}s")
    # 结果：时间相近
```

### 4.2 asyncio 事件循环原理

```text

│                                                             │
│  asyncio 事件循环工作原理：                                 │
│                                                             │

│  2. 循环监听 I/O 就绪事件                                  │
```

### 4.3 asyncio vs 多线程对比

```python
# 对比示例：并发请求 10 个 HTTP 接口

import asyncio
import aiohttp
import threading
import requests
import time

# asyncio 版本
async def fetch_all_asyncio(urls):
    async with aiohttp.ClientSession() as session:
        async def fetch(url):
            async with session.get(url) as response:
                return await response.text()
        
        tasks = [fetch(url) for url in urls]
        return await asyncio.gather(*tasks)

# 多线程版本
def fetch_all_threads(urls):
    def fetch(url):
        return requests.get(url).text
    
    threads = [threading.Thread(target=fetch, args=(url,)) for url in urls]
    for t in threads: t.start()
    for t in threads: t.join()

# 测试
urls = ["https://httpbin.org/delay/1"] * 10

start = time.time()
asyncio.run(fetch_all_asyncio(urls))
print(f"asyncio: {time.time() - start:.2f}s")  # ~1s

start = time.time()
fetch_all_threads(urls)
print(f"threads: {time.time() - start:.2f}s")  # ~2-3s（受 GIL 影响）
```

## 5. 解决方案

### 5.1 asyncio 正确用法

```python
# ✅ 正确的 asyncio 用法

import asyncio
import aiohttp
from typing import List

class AsyncAgentService:
    def __init__(self):
        self.session: aiohttp.ClientSession = None
        self.semaphore = asyncio.Semaphore(10)  # 限制并发数
    
    async def __aenter__(self):
        self.session = aiohttp.ClientSession(
            timeout=aiohttp.ClientTimeout(total=30)
        )
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self.session:
            await self.session.close()
    
    async def fetch_data(self, url: str) -> dict:
        """带超时的请求"""
        async with self.session.get(url) as response:
            return await response.json()
    
    async def batch_fetch(self, urls: List[str]) -> List[dict]:
        """并发请求，带限流"""
        async def limited_fetch(url):
            async with self.semaphore:  # 限流
                return await self.fetch_data(url)
        
        tasks = [limited_fetch(url) for url in urls]
        return await asyncio.gather(*tasks, return_exceptions=True)
    
    async def fetch_with_retry(self, url: str, max_retries: int = 3) -> dict:
        """重试机制"""
        for i in range(max_retries):
            try:
                return await self.fetch_data(url)
            except Exception as e:
                if i == max_retries - 1:
                    raise
                await asyncio.sleep(2 ** i)  # 指数退避
```

### 5.2 常见陷阱及解决方案

```python
# ❌ 陷阱 1：在协程中调用阻塞代码

async def bad_example():
    # 阻塞整个事件循环！
    result = blocking_function()  # 不要这样做！
    return result

async def good_example():
    # 使用 run_in_executor 包装
    result = await asyncio.get_event_loop().run_in_executor(
        None,  # 默认线程池
        blocking_function
    )
    return result

# ✅ 陷阱 2：忘记 await

async def bad_await():
    task = some_async_function()  # ❌ 没有 await
    return task  # 返回的是协程对象，不是结果

async def good_await():
    result = await some_async_function()  # ✅ 正确 await
    return result

# ✅ 陷阱 3：共享状态竞争

class Counter:
    def __init__(self):
        self.value = 0
        self._lock = asyncio.Lock()
    
    async def increment(self):
        async with self._lock:
            # 正确的异步锁
            self.value += 1

# ✅ 陷阱 4：嵌套事件循环

def outer_function():
    # 在同步函数中创建新事件循环
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)
    result = loop.run_until_complete(async_function())
    loop.close()
    return result

# ✅ 陷阱 5：会话管理

async def bad_session():
    """每次请求创建新会话"""
    async with aiohttp.ClientSession() as session:
        await session.get(url)
    # ❌ 连接池没有复用

async def good_session():
    """复用会话"""
    async with aiohttp.ClientSession() as session:
        # 在会话内执行多个请求
        await session.get(url1)
        await session.get(url2)
```

### 5.3 Agent 服务中的实践

```python
# Agent 服务架构示例

import asyncio
from abc import ABC, abstractmethod
from typing import Dict, Any, List
from dataclasses import dataclass

@dataclass
class AgentTask:
    id: str
    prompt: str
    tools: List[str]
    timeout: float = 30.0

class AsyncAgentRunner:
    """异步 Agent 执行器"""
    
    def __init__(self, max_concurrent: int = 10):
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.results: Dict[str, Any] = {}
    
    async def run_task(self, task: AgentTask) -> Dict[str, Any]:
        """执行单个任务"""
        async with self.semaphore:
            try:
                # 1. LLM 调用（异步）
                response = await self.call_llm(task.prompt)
                
                # 2. 解析工具调用
                tools_to_call = self.parse_tools(response)
                
                # 3. 并发执行工具
                tool_results = await self.execute_tools(tools_to_call)
                
                # 4. 整合结果
                return {
                    "task_id": task.id,
                    "response": response,
                    "tool_results": tool_results,
                    "status": "success"
                }
            except asyncio.TimeoutError:
                return {"task_id": task.id, "status": "timeout"}
            except Exception as e:
                return {"task_id": task.id, "status": "error", "error": str(e)}
    
    async def run_batch(self, tasks: List[AgentTask]) -> List[Dict[str, Any]]:
        """批量执行"""
        return await asyncio.gather(
            *[self.run_task(task) for task in tasks],
            return_exceptions=True
        )
    
    async def call_llm(self, prompt: str) -> str:
        """异步调用 LLM"""
        # 实际实现使用 aiohttp 调用 API
        await asyncio.sleep(0.1)  # 模拟
        return "LLM response"
    
    async def execute_tools(self, tools: List[str]) -> List[Dict]:
        """并发执行工具"""
        async def execute_single(tool: str):
            # 异步 RPC 调用
            await asyncio.sleep(0.05)
            return {"tool": tool, "result": "done"}
        
        return await asyncio.gather(
            *[execute_single(t) for t in tools],
            return_exceptions=True
        )
    
    def parse_tools(self, response: str) -> List[str]:
        """解析工具调用"""
        return ["tool1", "tool2"]
```

```python
# 核心修改逻辑：

# 1. 将同步代码改为异步
# 同步：
def fetch_data(url):
    return requests.get(url).json()

# 异步：
async def fetch_data(url):
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as resp:
            return await resp.json()

# 2. 添加超时和重试
async def fetch_with_timeout(url, timeout=5):
    try:
        return await asyncio.wait_for(fetch_data(url), timeout)
    except asyncio.TimeoutError:
        return None

# 3. 添加限流
semaphore = asyncio.Semaphore(10)
async def limited_fetch(url):
    async with semaphore:
        return await fetch_data(url)
```

## 9. 风险与注意事项

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
| 阻塞调用 | 事件循环卡住 | run_in_executor |

```python
        return "done"

    assert result is None  # 超时

    semaphore = asyncio.Semaphore(2)
    counter = {"value": 0}

        async with semaphore:
            counter["value"] += 1
            await asyncio.sleep(0.1)
    
    await asyncio.gather(*[task() for _ in range(5)])
    assert counter["value"] == 2  # 最多同时 2 个
```

## 11. 总结

1. **asyncio vs 多线程**：asyncio 适合 I/O 密集，多线程受 GIL 限制
2. **核心区别**：协程协作式调度 vs 线程抢占式调度
3. **Agent 服务**：推荐 asyncio + 进程池混合架构
4. **常见陷阱**：阻塞调用、未 await、共享状态
5. **最佳实践**：正确超时、限流、异常处理、资源管理

## 四、音乐缓存策略

### 4.1 分级缓存

```python
# QQ 音乐分级缓存策略
class MusicCache:
    def __init__(self):
        # L1: CDN 边缘缓存（高热点）
        self.l1_cache = LRUCache(capacity=1000, ttl=7*24*3600)
        
        # L2: 本地缓存（中热点）
        self.l2_cache = LRUCache(capacity=100, ttl=24*3600)
        
        # L3: 内存缓存（临时）
        self.l3_cache = LRUCache(capacity=10, ttl=300)
    
    def get(self, song_id):
        # L3 先查
        if song_id in self.l3_cache:
            return self.l3_cache.get(song_id)
        
        # L2 再查
        if song_id in self.l2_cache:
            data = self.l2_cache.get(song_id)
            self.l3_cache.set(song_id, data)
            return data
        
        # L1 最后查
        return self.fetch_from_cdn(song_id)
```

### 4.2 P2P 缓存机制

```python
# P2P 缓存贡献
class P2PCache:
    def __init__(self):
        self.max_storage = 1024 * 1024 * 100  # 100MB
        self.current_storage = 0
        self.cached_songs = {}
    
    def cache_song(self, song_id, data):
        """缓存歌曲并贡献给 P2P 网络"""
        size = len(data)
        
        # 检查空间
        if self.current_storage + size > self.max_storage:
            self.evict_lru()
        
        # 缓存
        self.cached_songs[song_id] = {
            'data': data,
            'size': size,
            'cached_at': time.time()
        }
        self.current_storage += size
        
        # 注册到 P2P 网络
        self.p2p_network.register(song_id, self.get_peers())
    
    def serve_peer(self, song_id, peer):
        """向其他 Peer 提供数据"""
        if song_id in self.cached_songs:
            return self.cached_songs[song_id]['data']
        return None
```

### 6.1 积分奖励

```python
# PCDN 贡献积分
class PCDNIncentive:
    def calculate_rewards(self, user_id,贡献量):
        # 1. 基础贡献（每 GB = 10 积分）
        base_score =贡献量 / (1024**3) * 10
        
        # 2. 在线奖励（每小时额外 5 积分）
        online_hours = self.get_online_time(user_id)
        online_bonus = online_hours * 5
        
        # 3. 热门奖励（贡献热门歌曲额外奖励）
        hot_bonus = 0
        for song_id in贡献的歌曲列表:
            if self.is_hot(song_id):
                hot_bonus +=贡献量 * 0.5
        
        # 4. 质量奖励（成功率加成）
        success_rate = self.get_success_rate(user_id)
        quality_multiplier = 1.0 + (success_rate - 0.9) * 2
        
        total = (base_score + online_bonus + hot_bonus) * quality_multiplier
        return total
    
    def settle(self, user_id, score):
        # 积分兑换
        exchange_rate = 0.01  # 1 积分 = 0.01 元
        cash = score * exchange_rate
        
        # 发放方式
        self.qq_wallet.deposit(user_id, cash)
```

## 七、技术实现

### 7.1 P2P 传输优化

```python
# 音乐流式传输
class MusicP2PStream:
    def __init__(self, song_id):
        self.song_id = song_id
        self.buffer = RingBuffer(capacity=30)  # 30秒缓冲
        self.pieces_received = 0
    
    def start_streaming(self):
        """边下载边播放"""
        # 1. 优先从 CDN/P2P 获取前 5 秒
        initial_data = self.fetch_initial_segment(5)
        self.play(initial_data)
        
        # 2. 后台继续下载后续内容
        self.background_download()
    
    def fetch_from_p2p(self, piece_id):
        """从 P2P 网络获取分片"""
        peers = self.get_peers_with_piece(piece_id)
        
        # 选择延迟最低的 peer
        best_peer = min(peers, key=lambda p: p.latency)
        
        return best_peer.request_piece(piece_id)
    
    def get_peers_with_piece(self, piece_id):
        """查找拥有指定分片的 peers"""
        # 1. 超级节点优先
        super_peers = self.p2p_network.get_super_peers(piece_id)
        if super_peers:
            return super_peers
        
        # 2. 普通节点
        return self.p2p_network.get_normal_peers(piece_id)
```

### 7.2 断点续传

```python
# 断点续传
class ResumeDownload:
    def __init__(self):
        self.checkpoints = {}
    
    def save_checkpoint(self, song_id, piece_id, offset):
        """保存下载进度"""
        self.checkpoints[song_id] = {
            'piece_id': piece_id,
            'offset': offset,
            'timestamp': time.time()
        }
        
        # 持久化到本地
        self.save_to_disk(song_id)
    
    def resume(self, song_id):
        """恢复下载"""
        checkpoint = self.load_checkpoint(song_id)
        
        if checkpoint:
            piece_id = checkpoint['piece_id']
            offset = checkpoint['offset']
            
            # 从断点继续下载
            return self.download_from_piece(song_id, piece_id, offset)
        
        return self.start_fresh(song_id)
```

## 九、面试常见问题

### Q1: PCDN 如何保证用户体验？

通过 CDN 兜底保证最低体验，P2P 作为加速补充；超级节点提供稳定高速源；激励机制保证节点质量；分级缓存策略优化命中。

### Q2: PCDN 的合规性如何处理？

用户参与需明确授权，积分奖励需符合当地法规；通过 VPN/加密通道保证传输安全；不缓存盗版内容。

### Q3: PCDN 的技术难点？

NAT 穿透成功率、节点稳定性、版权保护、防刷机制、激励公平性。

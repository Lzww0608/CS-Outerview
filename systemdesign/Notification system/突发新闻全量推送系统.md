## 三、消息分发策略

### 3.1 Kafka 广播模式

```python
import kafka

class NewsBroadcastService:
    def __init__(self):
        self.producer = kafka.KafkaProducer(
            bootstrap_servers=['kafka:9092'],
            compression_type='gzip'
        )
        
        # 创建广播 Topic
        self.create_broadcast_topic('breaking-news')
    
    def publish_breaking_news(self, news):
        """发布突发新闻"""
        message = {
            'news_id': news['id'],
            'title': news['title'],
            'content': news['content'][:200],  # 摘要
            'url': news['url'],
            'timestamp': time.time(),
            'priority': 'critical'
        }
        
        # 使用广播 Topic，所有消费者都能收到
        self.producer.send(
            'breaking-news',
            value=json.dumps(message),
            key='breaking'.encode()  # 固定 key 确保顺序
        )
        
        # 记录推送任务
        self.record_broadcast_task(news['id'])

# 创建广播消费者
class BroadcastConsumer:
    def __init__(self, consumer_group):
        self.consumer = kafka.KafkaConsumer(
            'breaking-news',
            group_id=consumer_group,  # 广播模式：每个 group 都收到
            bootstrap_servers=['kafka:9092']
        )
    
    def process(self):
        for message in self.consumer:
            news = json.loads(message.value)
            
            # 根据 consumer_group 分发不同任务
            if self.group_type == 'online':
                self.handle_online_users(news)
            elif self.group_type == 'offline':
                self.handle_offline_users(news)
```

import consistent_hash

class UserPartitionManager:
    def __init__(self, total_partitions=1000):
        self.total_partitions = total_partitions
        self.ring = consistent_hash.ConsistentHashRing()
        
        # 加载用户分区映射
        self.load_partition_map()

    def get_user_partition(self, user_id):

    def get_partition_users(self, partition_id):

        return self.db.query("""
            SELECT user_id, device_token, device_type, last_active_time
            FROM user_devices
            WHERE partition_id = %s
        """, [partition_id])

## 四、在线用户处理

### 4.1 长连接推送

```python
import asyncio

class WebSocketPushService:
    def __init__(self):
        self.connections = {}  # user_id -> WebSocket
    
    async def broadcast_to_online_users(self, news, user_partitions):
        """广播给所有在线用户"""
        # 1. 获取所有在线连接
        online_users = self.get_all_online_users()
        
        # 2. 过滤需要推送的用户
        target_users = [
            user_id for user_id in online_users
            if user_id in user_partitions
        ]
        
        # 3. 并行推送
        tasks = []
        for user_id in target_users:
            ws = self.connections.get(user_id)
            if ws and ws.open:
                tasks.append(self.send_to_user(ws, news))
        
        # 使用信号量控制并发
        semaphore = asyncio.Semaphore(10000)
        
        async def bounded_send(ws, news):
            async with semaphore:
                await ws.send_json(news)
        
        await asyncio.gather(*[
            bounded_send(ws, news) for ws in [self.connections[u] for u in target_users]
        ])
    
    async def send_to_user(self, ws, news):
        try:
            await ws.send_json({
                'type': 'breaking_news',
                'data': {
                    'news_id': news['news_id'],
                    'title': news['title'],
                    'content': news['content'],
                    'url': news['url'],
                    'timestamp': news['timestamp']
                }
            })
        except Exception as e:
            # 记录失败
            self.log_failed_push(news['news_id'], ws.user_id)
```

### 4.2 连接管理优化

```python
class ConnectionManager:
    def __init__(self):
        # 使用 Redis 存储连接状态
        self.redis = RedisCluster()
        
        # 分片连接池
        self.shard_pools = [ConnectionPool() for _ in range(16)]
    
    def register_connection(self, user_id, shard_id):
        """注册用户连接"""
        pool = self.shard_pools[shard_id % 16]
        pool.add_connection(user_id, self)
        
        # 记录在线状态
        self.redis.setex(f'online:{user_id}', 300, '1')
    
    def get_user_shard(self, user_id):
        """获取用户所属的分片"""
        return int(user_id) % 16
    
    async def broadcast_chunk(self, partition_id, news):
        """分片广播"""
        shard_id = partition_id % 16
        pool = self.shard_pools[shard_id]
        
        # 获取该分片的所有连接
        connections = pool.get_connections(partition_id)
        
        # 批量发送
        batch_size = 1000
        for i in range(0, len(connections), batch_size):
            batch = connections[i:i+batch_size]
            await asyncio.gather(*[
                conn.send(news) for conn in batch
            ])
```

## 五、离线用户处理

### 5.1 推送通道下发

```python
class OfflinePushService:
    def __init__(self):
        self.apns = APNsClient()
        self.fcm = FCMClient()
        self.tokens = TokenManager()
    
    def push_to_offline_users(self, news, partitions):
        """推送给离线用户"""
        # 1. 获取离线用户的设备 Token
        offline_devices = self.get_offline_devices(partitions)
        
        # 2. 按平台分组
        ios_tokens = []
        android_tokens = []
        
        for device in offline_devices:
            if device['device_type'] == 'ios':
                ios_tokens.append(device['token'])
            else:
                android_tokens.append(device['token'])
        
        # 3. 批量发送
        # iOS
        if ios_tokens:
            self.batch_push_ios(ios_tokens, news)
        
        # Android
        if android_tokens:
            self.batch_push_android(android_tokens, news)
    
    def batch_push_ios(self, tokens, news):
        """iOS 批量推送"""
        # APNs 批量限制：500 个 Token
        batch_size = 500
        
        for i in range(0, len(tokens), batch_size):
            batch = tokens[i:i+batch_size]
            
            payload = {
                'aps': {
                    'alert': {
                        'title': '突发新闻',
                        'body': news['title'],
                        'launch-image': news['image_url']
                    },
                    'sound': 'default',
                    'badge': 1
                },
                'news_id': news['news_id'],
                'url': news['url']
            }
            
            # 异步发送
            self.apns.send_batch_async(batch, payload)
```

### 5.2 设备 Token 管理

```python
class TokenManager:
    def __init__(self):
        self.redis = RedisCluster()
        self.db = Database()
    
    def get_offline_devices(self, partition_ids):
        """获取离线用户的设备列表"""
        # 查询最近 N 小时活跃的用户（视为可能在线但暂时离线）
        threshold = time.time() - 3600  # 1小时内
        
        results = self.db.query("""
            SELECT user_id, device_token, device_type
            FROM user_devices
            WHERE partition_id IN (%s)
            AND last_active_time < %s
            AND notification_enabled = 1
        """, [partition_ids, threshold])
        
        return results
    
    def refresh_tokens(self):
        """刷新无效 Token"""
        # APNs/FCM 反馈的无效 Token
        invalid_tokens = self.apns.get_feedback()
        
        for token_info in invalid_tokens:
            self.invalidate_token(token_info['token'])
            self.notify_user_to_update_token(token_info['user_id'])
```

### 6.2 推送合并

```python
class PushCoalescing:
    def __init__(self):
        self.pending_news = {}  # user_id -> [news_list]
        self.merge_window = 30  # 30秒合并窗口
    
    def add_news(self, user_id, news):
        """添加新闻到待推送队列"""
        if user_id not in self.pending_news:
            self.pending_news[user_id] = []
            
            # 启动定时器
            self.start_flush_timer(user_id, self.merge_window)
        
        self.pending_news[user_id].append(news)
    
    def flush_user(self, user_id):
        """合并推送"""
        news_list = self.pending_news.pop(user_id, [])
        if not news_list:
            return
        
        if len(news_list) == 1:
            # 单条推送
            self.push_service.push(user_id, news_list[0])
        else:
            # 合并推送
            merged = {
                'type': 'breaking_news_batch',
                'count': len(news_list),
                'items': news_list[:5],  # 最多显示5条
                'total': len(news_list)
            }
            self.push_service.push(user_id, merged)
```

class PushMonitor:
    def __init__(self):
        self.metrics = PrometheusClient()
    
    def record_metrics(self, news_id, stage, count, latency):
        """记录推送指标"""
        self.metrics.counter(
            'breaking_news_push_total',
            ['news_id', 'stage'],
            news_id, stage, count
        )
        
        self.metrics.histogram(
            'breaking_news_push_latency_seconds',
            ['stage'],
            stage, latency
        )
    
    def check_health(self):
        """健康检查"""
        # 推送成功率
        success_rate = self.get_success_rate()
        if success_rate < 0.95:
            self.alert('推送成功率过低', success_rate)
        
        # 推送延迟
        avg_latency = self.get_avg_latency()
        if avg_latency > 60:  # 60秒
            self.alert('推送延迟过高', avg_latency)

## 九、面试常见问题

### Q1: 如何保证亿级用户的推送实时性？

通过消息队列分片、用户分区并行处理、长连接优先推送、离线用户异步下发等多级策略。

### Q2: 如何避免推送风暴压垮系统？

1. 分区并行处理
2. 限流保护
3. 推送合并
4. 降级策略

### Q3: 如何区分在线和离线用户？

通过长连接状态判断在线；通过最后活跃时间判断可能离线；通过推送反馈确认实际状态。

### Q4: 突发新闻和普通推送的区别？

突发新闻优先级最高，需要动用所有资源；普通推送可延迟、可合并、有配额限制。

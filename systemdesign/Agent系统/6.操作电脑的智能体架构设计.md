# 操作电脑的智能体架构设计

## 1. 问题背景

能够操作电脑的智能体（Computer-Using Agent）是当前 AI 领域的热点方向。这类智能体可以像人类一样使用鼠标、键盘，操作桌面应用、浏览器、终端等。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  Computer-Using Agent 核心能力：                           │
│                                                             │
│  1. 感知屏幕                                              │
│     • 截图、OCR、UI 元素识别                               │
│     • 理解界面布局和状态                                   │
│                                                             │
│  2. 执行动作                                              │
│     • 鼠标点击、移动、拖拽                                │
│     • 键盘输入、快捷键                                    │
│     • 应用启动、文件操作                                   │
│                                                             │
│  3. 思考规划                                              │
│     • 理解任务目标                                        │
│     • 分解步骤                                            │
│     • 自我反思和纠错                                       │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

---

## 2. 问题现象

| 现象 | 描述 |
|------|------|
| 操作不精准 | 鼠标点击位置偏差 |
| 界面理解错误 | 无法识别 UI 元素 |
| 任务失败率高 | 长任务中途出错 |
| 效率低下 | 操作路径不优化 |

---

## 3. 影响范围

- 自动化办公（RPA 升级）
- 智能客服（桌面操作）
- 测试自动化（UI 测试）
- 数据采集（网页操作）
- 个人助手（任务自动化）

### 5.1 屏幕感知模块

```python
class ScreenPerception:
    """
    屏幕感知：捕获和理解屏幕内容
    """
    
    def __init__(self):
        self.screenshot_tool = screenshot
        self.ocr = PaddleOCR()
        self.ui_detector = UIElementDetector()
    
    def capture_screen(self):
        """截取屏幕"""
        return self.screenshot_tool.take()
    
    def detect_ui_elements(self, screenshot):
        """
        检测 UI 元素
        
        返回格式：
        [
            {"type": "button", "text": "确认", "bbox": [x1,y1,x2,y2], "confidence": 0.95},
            {"type": "input", "text": "用户名", "bbox": [x1,y1,x2,y2], "confidence": 0.88},
        ]
        """
        return self.ui_detector.detect(screenshot)
    
    def get_screen_state(self):
        """
        获取完整屏幕状态
        """
        screenshot = self.capture_screen()
        elements = self.detect_ui_elements(screenshot)
        text_content = self.ocr.extract_text(screenshot)
        
        return ScreenState(
            screenshot=screenshot,
            elements=elements,
            text=text_content,
            cursor_pos=get_cursor_position()
        )
```

### 5.3 输入执行模块

```python
class InputController:
    """
    输入控制：执行鼠标、键盘操作
    """
    
    def __init__(self):
        self.mouse = MouseController()
        self.keyboard = KeyboardController()
        self.screenshot = ScreenCapture()
    
    def click(self, x: int, y: int, button: str = "left"):
        """点击指定坐标"""
        self.mouse.move(x, y)
        self.mouse.click(button=button)
    
    def click_element(self, element: UIElement):
        """点击 UI 元素"""
        x, y = element.get_center()
        self.click(x, y)
    
    def type_text(self, text: str):
        """输入文本"""
        self.keyboard.type(text)
    
    def press_key(self, key: str):
        """按按键"""
        self.keyboard.press(key)
    
    def scroll(self, delta: int, x: int = None, y: int = None):
        """滚动"""
        if x and y:
            self.mouse.move(x, y)
        self.mouse.scroll(delta)
```

```python
class ToolSystem:
    """
    工具系统：管理可用的工具集
    """
    
    def __init__(self):
        self.tools = {
            # 基础操作
            'screenshot': self.take_screenshot,
            'click': self.click_element,
            'type': self.type_text,
            'press': self.press_key,
            'scroll': self.scroll,
            
            # 应用操作
            'open_app': self.open_application,
            'close_app': self.close_application,
            'switch_window': self.switch_window,
            
            # 文件操作
            'read_file': self.read_file,
            'write_file': self.write_file,
            'list_dir': self.list_directory,
            
            # 系统操作
            'run_command': self.run_terminal_command,
            'get_clipboard': self.get_clipboard,
            'set_clipboard': self.set_clipboard,
        }
    
    def execute(self, tool_name: str, **kwargs):
        """执行工具"""
        if tool_name not in self.tools:
            raise ValueError(f"Unknown tool: {tool_name}")
        
        return self.tools[tool_name](**kwargs)
    
    def get_available_tools(self):
        """获取可用工具列表"""
        return list(self.tools.keys())
```

### 5.5 完整 Agent 实现

```python
class ComputerUsingAgent:
    """
    操作电脑的智能体完整实现
    """
    
    def __init__(self, config: AgentConfig):
        # 核心组件
        self.perception = ScreenPerception()
        self.planner = TaskPlanner(config.llm)
        self.controller = InputController()
        self.tools = ToolSystem()
        self.memory = AgentMemory()
        
        # 配置
        self.max_steps = config.max_steps
        self.confidence_threshold = config.confidence_threshold
    
    def run(self, task: str) -> ExecutionResult:
        """
        执行任务
        
        Args:
            task: 用户任务描述
        
        Returns:
            ExecutionResult: 执行结果
        """
        self.memory.add_turn(role="user", content=task)
        
        for step in range(self.max_steps):
            # 1. 感知屏幕
            screen_state = self.perception.get_screen_state()
            
            # 2. 获取历史上下文
            context = self.memory.get_context()
            
            # 3. 规划下一步
            action = self.planner.plan_next_action(
                task=task,
                screen_state=screen_state,
                history=context
            )
            
            # 4. 执行动作
            result = self.execute_action(action)
            
            # 5. 检查结果
            if result.success:
                self.memory.add_turn(role="assistant", content=f"执行: {action}")
                
                if self.is_task_complete(result):
                    return ExecutionResult(success=True, message="任务完成")
            else:
                # 自我纠错
                correction = self.correct_error(action, result)
                if correction:
                    self.execute_action(correction)
        
        return ExecutionResult(success=False, message="任务超时")
    
    def execute_action(self, action: Action):
        """执行单个动作"""
        try:
            if action.type == 'tool':
                return self.tools.execute(action.name, **action.params)
            elif action.type == 'click':
                return self.controller.click_element(action.element)
            elif action.type == 'type':
                return self.controller.type_text(action.text)
            elif action.type == 'wait':
                time.sleep(action.duration)
                return Result(success=True)
        except Exception as e:
            return Result(success=False, error=str(e))
    
    def correct_error(self, action: Action, result: Result):
        """错误自我纠正"""
        correction_prompt = f"""
        上一步执行失败：
        动作：{action}
        错误：{result.error}
        当前屏幕：{self.perception.get_screen_state().describe()}
        
        请给出纠正方案：
        """
        correction = self.planner.get_correction(correction_prompt)
        return correction
```

## 9. 风险与注意事项

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
| 误操作 | 删除文件、发送错误信息 | 沙箱隔离、确认机制 |
| 循环执行 | Agent 卡死 | 最大步数限制 |
| UI 变化 | 检测失败 | 备用定位方式 |
| 隐私泄露 | 敏感信息被读取 | 数据脱敏、权限控制 |

## 10. 验证方式

```python
def benchmark_agent(agent: ComputerUsingAgent, test_cases: list):
    """
    智能体基准测试
    """
    results = []
    
    for case in test_cases:
        start_time = time.time()
        result = agent.run(case.task)
        duration = time.time() - start_time
        
        results.append({
            'task': case.task,
            'success': result.success,
            'duration': duration,
            'steps': result.steps,
            'error': result.error
        })
    
    # 计算指标
    success_rate = sum(r['success'] for r in results) / len(results)
    avg_duration = sum(r['duration'] for r in results) / len(results)
    
    return {
        'success_rate': success_rate,
        'avg_duration': avg_duration,
        'details': results
    }
```

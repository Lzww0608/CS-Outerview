# 21. 链上资产管理 AI Agent 平台系统设计

## 1. 问题背景

设计一个支持多链、多钱包、自然语言交互的链上资产管理 AI Agent 平台，让用户用自然语言管理跨链资产，降低 DeFi 使用门槛。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  平台愿景：                                                │
│  "把你的加密资产交给 AI 管家"                              │
│                                                             │
│  用户操作示例：                                            │
│  "把 ETH 换成 USDC，存进 Aave 赚利息"                     │
│  "查看我的资产配置，提醒我下周 Gas 低时调仓"                │
│  "发现 Uniswap 有套利机会，执行小额试单"                    │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

## 2. 问题现象

| 现象 | 描述 |
|------|------|
| 多链复杂性 | 主流链 10+，操作差异大 |

```python
# 意图解析器

class IntentParser:
    """自然语言意图解析"""
    
    def __init__(self, llm):
        self.llm = llm
        self.intent_schema = {
            "type": "object",
            "properties": {
                "action": {
                    "type": "string",
                    "enum": ["swap", "transfer", "deposit", "withdraw", "bridge", "stake"]
                },
                "source_token": {"type": "string"},
                "target_token": {"type": "string"},
                "amount": {"oneOf": [{"type": "string"}, {"type": "number"}]},
                "protocol": {"type": "string"},
                "chain": {"type": "string"},
                "constraints": {
                    "slippage": {"type": "number"},
                    "gas_limit": {"type": "number"},
                    "deadline": {"type": "number"}
                }
            },
            "required": ["action"]
        }
    
    async def parse(self, user_input: str, context: dict) -> Intent:
        """解析用户输入为结构化意图"""
        
        prompt = f"""
        用户输入："{user_input}"
        
        上下文：
        - 可用链：{context.get('supported_chains', [])}
        - 用户资产：{context.get('balances', {})}
        - 历史偏好：{context.get('preferences', {})}
        
        解析为 JSON：
        """
        
        response = await self.llm.generate(prompt)
        intent_data = json.loads(response)
        
        return Intent(
            action=intent_data["action"],
            source_token=intent_data.get("source_token"),
            target_token=intent_data.get("target_token"),
            amount=self._parse_amount(intent_data.get("amount")),
            protocol=intent_data.get("protocol"),
            chain=intent_data.get("chain"),
            constraints=intent_data.get("constraints", {})
        )
```

```python
# 多链适配器

class MultiChainAdapter:
    """多链统一适配器"""
    
    def __init__(self):
        self.chains = {
            "ethereum": EthereumAdapter(),
            "arbitrum": ArbitrumAdapter(),
            "polygon": PolygonAdapter(),
            # ... 更多链
        }
    
    async def execute(self, intent: Intent, wallet: Wallet) -> TransactionResult:
        adapter = self.chains.get(intent.chain)
        if not adapter:
            raise UnsupportedChainError(intent.chain)
        
        # 1. 获取报价
        quote = await adapter.get_quote(intent)
        
        # 2. 构建交易
        tx = await adapter.build_transaction(intent, quote, wallet)
        
        # 3. 签名
        signed_tx = await wallet.sign(tx)
        
        # 4. 发送
        return await adapter.send(signed_tx)
    
    async def bridge(self, from_chain: str, to_chain: str, 
                    token: str, amount: Decimal) -> TransactionResult:
        """跨链桥接"""
        bridge = self._select_bridge(from_chain, to_chain, token)
        return await bridge.execute(from_chain, to_chain, token, amount)
```

```python
# 钱包管理

class WalletManager:
    """统一钱包管理"""
    
    def __init__(self):
        self.wallets: Dict[str, Wallet] = {}
        self.mpc_coordinator = MPCCoordinator()
    
    async def create_mpc_wallet(self, owner: str, threshold: int = 2) -> MPCWallet:
        """创建 MPC 钱包"""
        # 生成密钥分片
        shards = await self.mpc_coordinator.generate_shards(threshold)
        
        wallet = MPCWallet(
            address=shards.address,
            shards=shards.shards,
            threshold=threshold
        )
        
        self.wallets[shards.address] = wallet
        return wallet
    
    async def sign(self, wallet_id: str, tx: Transaction) -> SignedTransaction:
        """MPC 签名"""
        wallet = self.wallets.get(wallet_id)
        if not isinstance(wallet, MPCWallet):
            raise InvalidWalletError(wallet_id)
        
        # 收集签名碎片
        signatures = await self.mpc_coordinator.collect_signatures(
            wallet.shards,
            tx.hash()
        )
        
        # 聚合签名
        return wallet.aggregate_signatures(signatures)
```

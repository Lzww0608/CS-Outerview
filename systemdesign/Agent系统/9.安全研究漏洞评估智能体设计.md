### 5.1 信息收集模块

```python
class ReconnaissanceModule:
    """
    信息收集模块
    """
    
    def __init__(self, target_ip, target_port):
        self.ip = target_ip
        self.port = target_port
        self.findings = {}
    
    def scan_ports(self):
        """端口扫描"""
        import socket
        
        # 快速检测端口开放
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(3)
            result = sock.connect_ex((self.ip, self.port))
            sock.close()
            
            if result == 0:
                self.findings['port_open'] = True
        except:
            self.findings['port_open'] = False
    
    def fingerprint_service(self):
        """服务指纹识别"""
        import socket
        
        # 尝试获取服务 banner
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            sock.connect((self.ip, self.port))
            sock.send(b"HEAD / HTTP/1.0\r\n\r\n")
            banner = sock.recv(1024)
            sock.close()
            
            self.findings['banner'] = banner.decode('utf-8', errors='ignore')
        except:
            self.findings['banner'] = None
    
    def detect_technologies(self):
        """技术栈识别"""
        # 基于 banner 和响应头识别技术
        technologies = []
        
        if 'Apache' in self.findings.get('banner', ''):
            technologies.append('Apache')
        if 'nginx' in self.findings.get('banner', '').lower():
            technologies.append('Nginx')
        if 'PHP' in self.findings.get('banner', ''):
            technologies.append('PHP')
        
        self.findings['technologies'] = technologies
```

```python
                'medium': sum(1 for v in vulnerabilities if v['severity'] == 'Medium'),
            },
            'vulnerabilities': vulnerabilities,
            'recommendations': self.generate_recommendations(vulnerabilities)
        }
        
        return report
    
    def generate_recommendations(self, vulnerabilities):
        """生成修复建议"""
        recommendations = []
        
        for vuln in vulnerabilities:
            rec = {
                'vulnerability': vuln['description'],
                'suggestion': self.get_fix_suggestion(vuln['type'])
            }
            recommendations.append(rec)
        
        return recommendations
    
    def get_fix_suggestion(self, vuln_type):
        """获取特定漏洞类型的修复建议"""
        suggestions = {
            'debug_mode': '关闭生产环境的调试模式',
            'error_leak': '配置统一的错误处理页面',
            'version_disclosed': '隐藏服务器版本信息',
        }
        return suggestions.get(vuln_type, '进行安全评估和修复')
```

### 5.4 完整安全研究智能体

```python
class SecurityResearchAgent:
    """
    安全研究智能体
    
    免责声明：
    本智能体仅用于授权的安全研究和教育目的
    使用前请确保：
    1. 拥有明确的授权许可
    2. 遵守适用法律法规
    3. 仅用于合法的安全测试场景
    """
    
    def __init__(self, target_ip, target_port):
        self.target_ip = target_ip
        self.target_port = target_port
        self.findings = {}
        self.vulnerabilities = []
    
    def run_assessment(self):
        """
        执行安全评估流程
        """
        # 1. 信息收集
        recon = ReconnaissanceModule(self.target_ip, self.target_port)
        recon.scan_ports()
        recon.fingerprint_service()
        recon.detect_technologies()
        self.findings = recon.findings
        
        # 2. 漏洞分析
        analyzer = VulnerabilityAnalysisModule(self.findings)
        analyzer.analyze_web_vulnerabilities()
        self.vulnerabilities = analyzer.vulnerabilities
        
        # 3. 生成报告
        report_gen = ReportGenerator()
        report = report_gen.generate_report(self.findings, self.vulnerabilities)
        
        return report
    
    def run(self):
        """执行评估并输出结果"""
        print(f"开始安全评估: {self.target_ip}:{self.target_port}")
        
        report = self.run_assessment()
        
        print(f"\n评估完成")
        print(f"发现 {report['summary']['total_vulnerabilities']} 个漏洞")
        
        return report
```

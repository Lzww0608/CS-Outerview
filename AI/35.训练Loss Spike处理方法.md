# 35. 训练中出现 Loss Spike 的处理方法

## 1. 问题背景

Loss Spike（损失突变）是大模型训练中的常见问题，指训练过程中 loss 突然剧烈上升或下降，破坏训练稳定性，影响模型收敛。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  正常训练曲线 vs 出现 Loss Spike：                         │
│                                                             │
│  Loss                                                       │
│    │                                                       │
│  2.5 ┤                    ┌──┐                             │
│      │                    │  │ ← Spike!                   │
│  2.0 ┤        ┌──────────┘  └──┐                        │
│      │        │                  │                         │
│  1.5 ┤────────┘                  └──────────             │
│      │                                                       │
│  1.0 ┼───────────────────────────────► Steps              │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

---

## 2. 问题现象

| 现象 | 描述 |
|------|------|
| Loss 突变上升 | Loss 突然飙升 10%-100%+ |
| Loss 突变下降 | Loss 突然骤降 |
| 训练震荡 | Loss 波动剧烈无法收敛 |
| NaN/Inf | Loss 变为非数值 |

---

## 3. 影响范围

- 所有深度学习训练任务
- 大模型预训练/SFT/RL
- 图像/语音/NLP 模型
- 分布式训练场景

## 5. 解决方案

### 5.1 学习率调整

```python
# 方案 1：梯度裁剪 + 学习率调整

import torch
from torch.nn.utils import clip_grad_norm_

class GradientClipper:
    """梯度裁剪防止爆炸"""
    
    def __init__(self, max_norm: float = 1.0):
        self.max_norm = max_norm
    
    def step(self, model):
        # 裁剪梯度范数
        clip_grad_norm_(model.parameters(), self.max_norm)

# 使用
clipper = GradientClipper(max_norm=1.0)

for batch in dataloader:
    optimizer.zero_grad()
    loss = model(batch)
    loss.backward()
    
    # 关键：在 optimizer.step() 之前裁剪
    clipper.step(model)
    optimizer.step()
```

```python
# 方案 2：学习率 warmup + decay

from torch.optim.lr_scheduler import CosineAnnealingWarmRestarts

class LearningRateScheduler:
    """学习率调度器"""
    
    def __init__(self, optimizer, warmup_steps: int, total_steps: int):
        self.optimizer = optimizer
        self.warmup_steps = warmup_steps
        self.total_steps = total_steps
        self.base_lr = optimizer.param_groups[0]['lr']
    
    def step(self, step: int):
        if step < self.warmup_steps:
            # Warmup 阶段：线性增长
            lr = self.base_lr * (step / self.warmup_steps)
        else:
            # 余弦退火
            progress = (step - self.warmup_steps) / (self.total_steps - self.warmup_steps)
            lr = self.base_lr * 0.5 * (1 + math.cos(math.pi * progress))
        
        for param_group in self.optimizer.param_groups:
            param_group['lr'] = lr
        
        return lr
```

### 5.2 异常数据过滤

```python
# 方案 3：数据清洗与过滤

class DataFilter:
    """过滤异常训练数据"""
    
    def __init__(self, max_token_len: int = 2048, 
                 max_loss_threshold: float = 5.0):
        self.max_token_len = max_token_len
        self.max_loss_threshold = max_loss_threshold
    
    def filter_batch(self, batch: dict) -> bool:
        # 检查 token 长度
        if batch.get('input_ids', torch.tensor([])).shape[-1] > self.max_token_len:
            return False
        
        # 检查标签是否有效
        if batch.get('labels', torch.tensor([])).max() < 0:
            return False
        
        return True
    
    def detect_outlier_loss(self, losses: list, window: int = 100) -> bool:
        """检测 loss 是否异常"""
        if len(losses) < window:
            return False
        
        recent_losses = losses[-window:]
        mean = sum(recent_losses) / len(recent_losses)
        std = (sum((x - mean) ** 2 for x in recent_losses) / len(recent_losses)) ** 0.5
        
        # 如果当前 loss 超过 mean + 3*std，视为异常
        current_loss = losses[-1]
        if abs(current_loss - mean) > 3 * std:
            return True
        
        return False
```

### 5.3 混合精度训练配置

```python
# 方案 4：正确的混合精度训练

from torch.cuda.amp import autocast, GradScaler

class StableTrainer:
    """稳定训练器"""
    
    def __init__(self, model, optimizer, init_scale: float = 2**16):
        self.model = model
        self.optimizer = optimizer
        self.scaler = GradScaler(init_scale=init_scale)
    
    def train_step(self, batch):
        with autocast(dtype=torch.float16):
            outputs = self.model(**batch)
            loss = outputs.loss
        
        # 检查 loss 是否有效
        if torch.isnan(loss) or torch.isinf(loss):
            print(f"⚠️ Invalid loss detected: {loss}")
            return None
        
        # 梯度缩放
        self.scaler.scale(loss).backward()
        
        # 梯度裁剪
        self.scaler.unscale_(self.optimizer)
        torch.nn.utils.clip_grad_norm_(self.model.parameters(), 1.0)
        
        # 更新权重
        self.scaler.step(self.optimizer)
        self.scaler.update()
        self.optimizer.zero_grad()
        
        return loss.item()
    
    def detect_nan_gradients(self):
        """检测梯度 NaN"""
        for name, param in self.model.named_parameters():
            if param.grad is not None:
                if torch.isnan(param.grad).any():
                    print(f"⚠️ NaN gradient in {name}")
                    return True
        return False
```

### 5.4 训练恢复机制

```python
# 方案 5：自动检测与恢复

import os
import json

class TrainingRecovery:
    """训练恢复机制"""
    
    def __init__(self, checkpoint_dir: str, spike_threshold: float = 2.0):
        self.checkpoint_dir = checkpoint_dir
        self.spike_threshold = spike_threshold  # loss 突增倍数
        self.history = []
        self.best_loss = float('inf')
    
    def check_and_handle_spike(self, step: int, loss: float) -> bool:
        """
        检测 Spike 并处理
        返回 True 表示需要回滚
        """
        self.history.append({'step': step, 'loss': loss})
        
        if loss < self.best_loss * 0.9:  # loss 显著下降
            self.best_loss = loss
            self.save_checkpoint(step, 'best')
        
        # 检测 Spike
        if len(self.history) >= 50:
            recent_losses = [h['loss'] for h in self.history[-50:]]
            mean = sum(recent_losses) / len(recent_losses)
            
            # 如果当前 loss 是均值的 spike_threshold 倍
            if loss > mean * self.spike_threshold:
                print(f"⚠️ Spike detected: loss={loss}, mean={mean}")
                return True
        
        return False
    
    def rollback_to_best(self, model, optimizer) -> int:
        """回滚到最佳状态"""
        checkpoint_path = os.path.join(self.checkpoint_dir, 'best.pt')
        
        if os.path.exists(checkpoint_path):
            checkpoint = torch.load(checkpoint_path)
            model.load_state_dict(checkpoint['model_state_dict'])
            optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
            
            # 跳过导致 Spike 的数据
            return checkpoint.get('next_step', 0)
        
        return 0
    
    def save_checkpoint(self, step: int, tag: str):
        """保存检查点"""
        os.makedirs(self.checkpoint_dir, exist_ok=True)
        
        checkpoint = {
            'step': step,
            'model_state_dict': self.model.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'history': self.history
        }
        
        path = os.path.join(self.checkpoint_dir, f'{tag}.pt')
        torch.save(checkpoint, path)
```

## 6. 可选方案对比

| 方案 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| 梯度裁剪 | 简单有效 | 保守 | 通用 |
| 学习率调整 | 根本解决 | 需调参 | 学习率问题 |
| 数据过滤 | 预防为主 | 可能丢失信息 | 数据问题 |
| 混合精度 | 加速 + 稳定 | 精度损失 | 大模型 |
| 自动回滚 | 自动化程度高 | 实现复杂 | 生产环境 |

## 9. 风险与注意事项

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
| 过度裁剪 | 收敛慢 | 设置合理的 max_norm |
| 数据过滤过度 | 丢失信息 | 保守过滤 |
| 回滚丢失进度 | 训练时间 | 保存中间检查点 |
| 精度损失 | 模型效果 | 使用 BF16 而非 FP16 |

## 11. 总结

1. **Loss Spike 本质**：梯度爆炸 + 数据异常 + 配置不当的综合结果
2. **核心解决思路**：梯度裁剪 + 学习率调度 + 数据过滤 + 自动恢复
3. **预防为主**：数据清洗 + 合理配置 + 监控告警
4. **处理策略**：检测到 Spike 后回滚 + 调整配置 + 跳过异常数据
5. **最佳实践**：完整配置 + 定期检查点 + 异常监控

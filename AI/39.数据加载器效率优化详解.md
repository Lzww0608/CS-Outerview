# 39. 数据加载器效率优化详解

## 1. 问题背景

数据加载效率是训练 pipeline 的关键瓶颈。当 GPU 计算速度超过数据加载速度时，GPU 会处于空闲等待状态，造成硬件资源浪费。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  训练 Pipeline 流程：                                     │
│                                                             │
│  ┌─────────┐    ┌─────────┐    ┌─────────┐    ┌───────┐ │
│  │ Dataset │───►│ DataLoader│───►│  GPU   │───►│ Opt. │ │
│  │         │    │          │    │  Train │    │ Step  │ │
│  └─────────┘    └─────────┘    └─────────┘    └───────┘ │
│       ▲              ▲              ▲                         │
│       │              │              │                         │
│  数据读取      批处理+      计算+反向            │
│  + Tokenize    collate        传播                           │
│                                                             │
│  瓶颈：DataLoader 往往成为最慢的一环                        │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

---

## 2. 问题现象

| 现象 | 描述 |
|------|------|
| GPU 利用率低 | 数据加载跟不上计算 |
| 训练时间长 | 每 epoch 耗时远超预期 |
| Workers 空闲 | num_workers 设置不当 |
| 内存爆炸 | prefetch_factor 过大 |

---

## 3. 影响范围

- 所有深度学习训练任务
- 大规模预训练
- 分布式训练
- 多模态训练

### 5.2 数据预处理优化

```python
# 方案 1：预计算 Tokenize

class PreTokenizedDataset(Dataset):
    """预分词数据集，避免训练时 tokenize"""
    
    def __init__(self, raw_text_file, tokenizer):
        # 训练前预先分词
        with open(raw_text_file, 'r') as f:
            texts = f.readlines()
        
        # 批量 tokenize
        self.tokenized = tokenizer(
            texts,
            padding=True,
            truncation=True,
            max_length=2048,
            return_tensors='pt'
        )
        
        # 释放原始文本
        del texts
    
    def __len__(self):
        return self.tokenized['input_ids'].shape[0]
    
    def __getitem__(self, idx):
        return {
            'input_ids': self.tokenized['input_ids'][idx],
            'attention_mask': self.tokenized['attention_mask'][idx],
        }

# 方案 2：Memory-mapped 文件

class MMapDataset(Dataset):
    """使用内存映射加速读取"""
    
    def __init__(self, file_path):
        # .bin 文件已包含序列化的 token IDs
        self.data = np.memmap(file_path, dtype=np.uint16, mode='r')
    
    def __len__(self):
        return len(self.data) // seq_len
    
    def __getitem__(self, idx):
        start = idx * seq_len
        return torch.from_numpy(self.data[start:start+seq_len].copy())
```

### 5.3 并行预处理

```python
# 方案 3：多进程 Tokenize

from concurrent.futures import ProcessPoolExecutor
from functools import partial

def tokenize_text(text, tokenizer):
    return tokenizer(text, truncation=True, max_length=2048)

def parallel_tokenize(file_list, tokenizer, num_workers=8):
    """多进程 tokenize"""
    with ProcessPoolExecutor(max_workers=num_workers) as executor:
        tokenize_fn = partial(tokenize_text, tokenizer=tokenizer)
        results = list(executor.map(tokenize_fn, file_list))
    return results

# 方案 4：使用 Arrow/Parquet

import pyarrow.parquet as pq

class ParquetDataset(Dataset):
    """使用 Parquet 加速读取"""
    
    def __init__(self, parquet_file):
        self.parquet_file = pq.ParquetFile(parquet_file)
        self.columns = ['input_ids', 'attention_mask', 'labels']
    
    def __len__(self):
        return self.parquet_file.metadata.num_rows
    
    def __getitem__(self, idx):
        # Parquet 按列存储，读取高效
        table = self.parquet_file.read_row_group(
            idx // rows_per_group,
            columns=self.columns
        )
        return {
            'input_ids': torch.tensor(table['input_ids'][idx % rows_per_group]),
            'attention_mask': torch.tensor(table['attention_mask'][...]),
        }
```

### 5.4 自定义高效 Collate

```python
# 方案 5：优化 Collate 函数

def optimized_collate_fn(batch):
    """优化的 collate 函数"""
    
    # 直接从 batch 中提取
    input_ids = torch.stack([x['input_ids'] for x in batch])
    attention_mask = torch.stack([x['attention_mask'] for x in batch])
    
    # 处理 labels（可能不存在）
    if 'labels' in batch[0]:
        labels = torch.stack([x.get('labels', -100) for x in batch])
    else:
        labels = None
    
    return {
        'input_ids': input_ids,
        'attention_mask': attention_mask,
        'labels': labels,
    }

# 动态 Padding

def dynamic_padding_collate(batch, max_length=None):
    """动态 padding 到 batch 内最大长度"""
    
    input_ids = [x['input_ids'] for x in batch]
    attention_mask = [x['attention_mask'] for x in batch]
    
    # 找到 batch 内最大长度
    if max_length is None:
        max_len = max(len(ids) for ids in input_ids)
    else:
        max_len = min(max(len(ids) for ids in input_ids), max_length)
    
    # Padding
    padded_input_ids = []
    padded_attention_mask = []
    
    for ids, mask in zip(input_ids, attention_mask):
        pad_len = max_len - len(ids)
        padded_input_ids.append(
            torch.cat([ids, torch.zeros(pad_len, dtype=torch.long)])
        )
        padded_attention_mask.append(
            torch.cat([mask, torch.zeros(pad_len, dtype=torch.long)])
        )
    
    return {
        'input_ids': torch.stack(padded_input_ids),
        'attention_mask': torch.stack(padded_attention_mask),
    }
```

### 5.5 GPU 直接读取

```python
# 方案 6：GPU 直接加载

class GPUDirectDataset(Dataset):
    """数据直接加载到 GPU"""
    
    def __init__(self, data_path, device='cuda'):
        # 预加载到 GPU 内存
        data = torch.load(data_path)
        self.input_ids = data['input_ids'].to(device)
        self.attention_mask = data['attention_mask'].to(device)
        self.device = device
    
    def __getitem__(self, idx):
        return {
            'input_ids': self.input_ids[idx],
            'attention_mask': self.attention_mask[idx],
        }
```

## 6. 可选方案对比

| 方案 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| 增加 num_workers | 简单 | CPU 资源占用 | 通用 |
| 预分词 | 最快 | 预处理耗时 | 固定长度 |
| MMap | 内存高效 | 随机访问慢 | 超大数据集 |
| Parquet | 列式读取快 | 需转换格式 | 大数据分析 |
| GPU Direct | 最快 | 显存占用大 | 小数据集 |

### 7.2 GPU/CPU 重叠

```python
# GPU/CPU 重叠实现

from torch.cuda.streams import Stream

class OverlappedDataLoader:
    """重叠数据加载和 GPU 计算"""
    
    def __init__(self, dataloader, device='cuda'):
        self.dataloader = dataloader
        self.stream = Stream(device)
        self.device = device
        self.next_batch = None
    
    def __iter__(self):
        # 预加载第一批
        self.next_batch = next(iter(self.dataloader))
        
        for batch in self.dataloader:
            # 当前 batch 移到 GPU
            with torch.cuda.stream(self.stream):
                current_batch = {k: v.to(self.device) for k, v in self.next_batch.items()}
            
            # 预加载下一批（主 stream）
            try:
                self.next_batch = next(self.dataloader_iter)
            except StopIteration:
                self.next_batch = None
            
            yield current_batch
    
    def __len__(self):
        return len(self.dataloader)
```

## 11. 总结

1. **核心瓶颈**：I/O、Tokenize、Collate 是主要瓶颈
2. **基础优化**：num_workers + pin_memory + persistent_workers
3. **格式优化**：二进制格式 > JSON/CSV
4. **高级优化**：GPU/CPU 重叠、预分词
5. **最佳实践**：测量 → 针对性优化 → 验证效果

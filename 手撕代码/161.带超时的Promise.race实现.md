# 161. 带超时的 Promise.race 实现

## 题目

实现一个 `promiseRaceWithTimeout`，要求：
1. 接收一个 Promise 数组和一个超时时间
2. 返回最快成功解析的 Promise 结果
3. 如果超时则抛出超时错误
4. 其他未完成的 Promise 应当被正确清理

## 问题背景

标准 `Promise.race()` 的问题在于：
- 一旦有 Promise 率先 settle，后续 Promise 仍可能继续执行
- 没有超时机制，可能导致无限等待
- 无法取消已启动但尚未完成的任务

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  Promise.race 的局限：                                     │
│                                                             │
│  Promise.race([fetchA(), fetchB(), fetchC()])              │
│       │                                                     │
│       ▼                                                     │
│  ┌─────────────────────────────────────────────────────┐  │
│  │ A 率先 resolve ──► 返回结果                         │  │
│  │ B 和 C 仍在执行 ──► 无法取消，继续消耗资源          │  │
│  └─────────────────────────────────────────────────────┘  │
│                                                             │
│  需求：超时控制 + 资源清理                                │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

## 原因分析

### 标准 Promise.race 缺陷

```javascript
// 标准 Promise.race 的问题

Promise.race([
    fetch('/api/slow'),
    fetch('/api/fast')
])
    .then(result => console.log(result))
    .catch(err => console.error(err));

// 问题：
// 1. 即使 fetch('/api/fast') 先返回，/api/slow 仍在后台执行
// 2. 无法设置超时时间
// 3. 无法取消失败的请求
```

### 理想行为

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  理想行为：                                                │
│                                                             │
│  ┌─────────────────────────────────────────────────────┐  │
│  │ promiseRaceWithTimeout                             │  │
│  │   .promises([A, B, C])                            │  │
│  │   .timeout(2000)                                  │  │

│  • A 先成功 ──► 返回 A 的结果                            │

│                                                             │
```

## 解决方案

### 方案一：基础版

```javascript
/**
 * 基础版：Promise.race + Promise + 超时
 * 
 * 优点：实现简单
 * 缺点：无法取消未完成的 Promise
 */
function promiseRaceWithTimeout(promises, timeoutMs, timeoutError = new Error('Timeout')) {
    // 创建超时 Promise
    const timeoutPromise = new Promise((_, reject) => {
        setTimeout(() => reject(timeoutError), timeoutMs);
    });
    
    // race 竞速
    return Promise.race([...promises, timeoutPromise]);
}

// 使用示例
const result = await promiseRaceWithTimeout(
    [
        fetch('/api/v1'),  // 慢
        fetch('/api/v2'),  // 快
        fetch('/api/v3')   // 中
    ],
    2000  // 2秒超时
);
```

### 方案二：可取消版（使用 AbortController）

```javascript
/**
 * 可取消版：使用 AbortController
 * 
 * 优点：可以取消进行中的请求
 * 缺点：需要目标 Promise 支持 AbortSignal
 */
function promiseRaceWithAbort(promises, timeoutMs, timeoutError = new Error('Timeout')) {
    // 创建 AbortController
    const controller = new AbortController();
    const { signal } = controller;
    
    // 包装 Promise，添加取消支持
    const abortablePromises = promises.map(p => {
        // 如果是 fetch 等支持 signal 的 API
        if (p instanceof Request || p instanceof URL) {
            return fetch(p, { signal });
        }
        
        // 通用包装：添加 .cancel()
        const wrapped = Promise.race([
            p,
            new Promise((_, reject) => 
                signal.addEventListener('abort', () => reject(new Error('Cancelled')))
            )
        ]);
        
        wrapped.cancel = () => controller.abort();
        return wrapped;
    });
    
    // 超时控制器
    const timeoutController = new AbortController();
    const timeoutId = setTimeout(() => timeoutController.abort(), timeoutMs);
    
    return Promise.race([
        Promise.race(abortablePromises),
        new Promise((_, reject) => 
            timeoutController.signal.addEventListener('abort', () => {
                clearTimeout(timeoutId);
                // 取消所有未完成的 Promise
                abortablePromises.forEach(p => p.cancel?.());
                reject(timeoutError);
            })
        )
    ]);
}
```

### 方案三：完整版（返回元组）

```javascript
/**
 * 完整版：返回结果和状态信息
 * 
 * 返回格式：
 * {
 *   result: T | null,           // 结果或 null
 *   status: 'resolved' | 'timeout' | 'rejected',
 *   winnerIndex: number,        // 胜出的 Promise 索引
 *   pendingPromises: number[],  // 未完成的 Promise 索引
 *   elapsedTime: number         // 实际耗时
 * }
 */
async function promiseRaceWithTimeout({
    promises,
    timeoutMs,
    timeoutError = new Error('Timeout'),
    onTimeout = null,
    onSettled = null
}) {
    const startTime = Date.now();
    const results = new Array(promises.length).fill(undefined);
    let winnerIndex = -1;
    let status = 'timeout';
    
    // 创建 Deferred Promise 用于手动 resolve
    const deferred = {};
    deferred.promise = new Promise((resolve, reject) => {
        deferred.resolve = resolve;
        deferred.reject = reject;
    });
    
    // 为每个 Promise 添加回调
    const trackedPromises = promises.map((p, index) => {
        return Promise.resolve(p).then(
            (value) => {
                if (winnerIndex === -1) {
                    winnerIndex = index;
                    status = 'resolved';
                    results[index] = { success: true, value };
                    deferred.resolve({
                        result: value,
                        status,
                        winnerIndex,
                        pendingPromises: getPendingIndices(index),
                        elapsedTime: Date.now() - startTime
                    });
                }
            },
            (error) => {
                if (winnerIndex === -1) {
                    results[index] = { success: false, error };
                    // 只有一个 reject 时不立即 resolve，等超时
                }
            }
        );
    });
    
    // 超时处理
    const timeoutPromise = new Promise((resolve) => {
        setTimeout(() => {
            if (winnerIndex === -1) {
                status = 'timeout';
                const pendingIndices = results
                    .map((r, i) => r === undefined ? i : -1)
                    .filter(i => i !== -1);
                
                onTimeout?.(pendingIndices);
                
                resolve({
                    result: null,
                    status,
                    winnerIndex: -1,
                    pendingPromises: pendingIndices,
                    elapsedTime: Date.now() - startTime
                });
            }
        }, timeoutMs);
    });
    
    // 注册完成回调
    const settledPromise = Promise.all(trackedPromises).then(() => {
        onSettled?.(results);
    });
    
    return Promise.race([deferred.promise, timeoutPromise]);
}

// 辅助函数
function getPendingIndices(winnerIndex) {
    return []; // 在实际使用时需要追踪
}
```

### 方案四：Generator/Async Iterator 版

```javascript
/**
 * Generator 版：更灵活的控制流
 */
function* createRacingPromises(promises, timeoutMs) {
    const controllers = promises.map(() => new AbortController());
    
    try {
        // 并发执行
        const results = await Promise.race([
            Promise.all(
                promises.map((p, i) => 
                    Promise.resolve(p).finally(() => controllers[i].abort())
                )
            ),
            new Promise((_, reject) => 
                setTimeout(() => reject(new Error('Timeout')), timeoutMs)
            )
        ]);
        
        // 找到成功的结果
        const winnerIndex = results.findIndex(r => r !== undefined);
        yield { type: 'resolved', index: winnerIndex, value: results[winnerIndex] };
        
    } catch (error) {
        yield { type: 'timeout', error };
        
    } finally {
        // 清理：取消所有未完成的控制器
        controllers.forEach(c => c.abort());
    }
}

// 使用
for await (const event of createRacingPromises([
    fetchA(),
    fetchB(),
    fetchC()
], 2000)) {
    if (event.type === 'resolved') {
        console.log(`Winner: ${event.index}`, event.value);
        break;
    }
}
```

## 推荐方案

### 场景一：简单 HTTP 请求

```javascript
// 推荐：AbortController + fetch
async function fetchWithTimeout(url, options = {}) {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), options.timeout || 5000);
    
    try {
        return await fetch(url, {
            ...options,
            signal: controller.signal
        });
    } finally {
        clearTimeout(timeoutId);
    }
}

// 使用
const result = await Promise.race([
    fetchWithTimeout('/api/fast'),
    fetchWithTimeout('/api/slow')
]);
```

### 场景二：通用 Promise（不支持 Abort）

```javascript
// 推荐：包装 + 清理
class CancellablePromise {
    constructor(promise) {
        this.promise = promise;
        this.cancelled = false;
    }
    
    cancel() {
        this.cancelled = true;
    }
    
    then(onFulfilled, onRejected) {
        return this.promise.then(
            (v) => this.cancelled ? onRejected('Cancelled') : onFulfilled(v),
            (e) => this.cancelled ? onRejected('Cancelled') : onRejected(e)
        );
    }
}

async function promiseRaceWithTimeout(promises, timeoutMs) {
    const wrapped = promises.map(p => new CancellablePromise(p));
    
    return Promise.race([
        Promise.race(wrapped),
        new Promise((_, reject) => 
            setTimeout(() => {
                wrapped.forEach(p => p.cancel());
                reject(new Error('Timeout'));
            }, timeoutMs)
        )
    ]);
}
```

## 实施步骤

### 核心逻辑

```javascript
// 核心修改逻辑：

// 1. 创建超时控制器
const timeoutController = new AbortController();

// 2. 设置超时
const timeoutId = setTimeout(() => timeoutController.abort(), timeoutMs);

// 3. 添加到 Promise.race
return Promise.race([
    ...promises,
    new Promise((_, reject) => 
        timeoutController.signal.addEventListener('abort', () => reject(error))
    )
]);

// 4. 清理
clearTimeout(timeoutId);
```

## 风险与注意事项

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
| Abort 不支持 | Promise 继续执行 | 使用 AbortController polyfill |
| 内存泄漏 | 闭包持有引用 | 显式清理 |
| 竞态条件 | 结果不确定 | 使用 winnerIndex |
| 超时清理 | 资源未释放 | finally 块清理 |

```javascript
            new Promise((r) => {
                setTimeout(() => {

                }, 100);
            })
        ];

        await new Promise(r => setTimeout(r, 200));
        // 验证 cancelled 状态
    });
```

## 总结

1. **基础实现**：Promise.race + setTimeout
2. **可取消**：AbortController + signal
3. **完整版**：返回元组 + 状态信息
4. **清理**：finally 块清理资源
5. **最佳实践**：根据场景选择方案，优先使用 AbortController

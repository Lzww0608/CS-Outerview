class Solution {
    public:
        // 定义输入字符的类型
        enum InputType {
            INVALID,    // 0
            SPACE,      // 1 (题目通常要求去除首尾空格，若中间有空格则非法，这里简化处理，假设预处理已去除首尾)
            SIGN,       // 2
            DIGIT,      // 3
            DOT,        // 4
            EXPONENT    // 5
        };
    
        // 辅助函数：将字符映射为 InputType
        InputType getCharType(char c) {
            if (std::isdigit(c)) return DIGIT;
            if (c == '+' || c == '-') return SIGN;
            if (c == '.') return DOT;
            if (c == 'e' || c == 'E') return EXPONENT;
            return INVALID;
        }
    
        bool isNumber(string s) {
            // 定义状态转移矩阵
            // 行索引代表当前状态 (0-8)
            // 列索引代表输入类型: [INVALID, SPACE, SIGN, DIGIT, DOT, EXPONENT]
            // 值代表下一个状态，-1 表示非法转移
            
            // 状态定义：
            // 0: Initial (初始)
            // 1: Sign (符号位)
            // 2: Integer (整数数字) -> 合法终态
            // 3: Point w/o Int (前面无数字的小数点，如 .5)
            // 4: Point w/ Int (前面有数字的小数点，如 3.) -> 合法终态
            // 5: Fraction (小数部分数字) -> 合法终态
            // 6: Exp (指数符号 e/E)
            // 7: Exp Sign (指数后的符号)
            // 8: Exp Number (指数后的数字) -> 合法终态
            
            // 映射关系：INVALID=0, SPACE=1 (暂不处理), SIGN=2, DIGIT=3, DOT=4, EXP=5
            const int transfer[9][6] = {
                // INV, SP, SGN, DIG, DOT, EXP
                { -1, -1,  1,   2,   3,  -1 }, // State 0: Initial
                { -1, -1, -1,   2,   3,  -1 }, // State 1: Sign
                { -1, -1, -1,   2,   4,   6 }, // State 2: Integer (Valid End)
                { -1, -1, -1,   5,  -1,  -1 }, // State 3: Point w/o Int
                { -1, -1, -1,   5,  -1,   6 }, // State 4: Point w/ Int (Valid End)
                { -1, -1, -1,   5,  -1,   6 }, // State 5: Fraction (Valid End)
                { -1, -1,  7,   8,  -1,  -1 }, // State 6: Exp
                { -1, -1, -1,   8,  -1,  -1 }, // State 7: Exp Sign
                { -1, -1, -1,   8,  -1,  -1 }  // State 8: Exp Number (Valid End)
            };
    
            int state = 0; // 从初始状态开始
    
            for (char c : s) {
                InputType type = getCharType(c);
                
                // 如果字符非法，直接返回 false
                if (type == INVALID) return false;
                
                // 获取下一个状态
                state = transfer[state][type];
                
                // 如果转移非法，返回 false
                if (state == -1) return false;
            }
    
            // 检查最终状态是否为合法终态
            // 合法终态有: 2 (整数), 4 (数字加点), 5 (小数), 8 (指数数字)
            return state == 2 || state == 4 || state == 5 || state == 8;
        }

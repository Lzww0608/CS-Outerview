```cpp
/*
在广告的某些场景中，需要提前请求后台
将广告信息缓存在终端本地，然后在需要
展示广告的时候直接从终端本地缓存中选
择1个广告进行展示。同时，由于终端的
存储空间有限，需要限制缓存的广告数量
上限。请针对该场景，设计一个数据结构
用于缓存广告，要求如下：
1、提供插入的接口，并且确保缓存的数
量不会突破上限N；如果超过上限，则将
最早插入的广告淘汰；
2、提供查询的接口，每个广告有一个对
应的价值（浮点数），希望该接口返回价
值最大的广告
3、两个接口的时间复杂度应当尽量低，
其中查询接口的时间复杂度要求更高
*/
#include <queue>
#include <deque>
#include <string>
#include <stdexcept>
using namespace std;

struct Ad {
    long long id;
    double value;
    string content;
};

class AdCache {
private:
    int capacity;
    long long nextId = 0;
    queue<Ad> q;
    deque<Ad> maxDeque;

public:
    AdCache(int N) : capacity(N) {}

    void insert(double value, const string& content) {
        Ad ad{nextId++, value, content};

        // 1. 插入主队列
        q.push(ad);

        // 2. 维护单调递减队列
        while (!maxDeque.empty() && maxDeque.back().value <= ad.value) {
            maxDeque.pop_back();
        }
        maxDeque.push_back(ad);

        // 3. 超过容量则淘汰最老广告
        if ((int)q.size() > capacity) {
            Ad old = q.front();
            q.pop();

            if (!maxDeque.empty() && maxDeque.front().id == old.id) {
                maxDeque.pop_front();
            }
        }
    }

    Ad getMaxAd() const {
        if (maxDeque.empty()) {
            throw runtime_error("cache is empty");
        }
        return maxDeque.front();
    }

    bool empty() const {
        return q.empty();
    }

    int size() const {
        return (int)q.size();
    }
};
```

# 智能合约详解：从概念到实战

## 一、智能合约概述

### 1.1 什么是智能合约

智能合约（Smart Contract）是部署在区块链上的可执行程序，当满足预设条件时自动触发执行。Nick Szabo 于 1994 年首次提出这一概念，比特币和以太坊使其成为现实。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  智能合约核心特性：                                         │
│                                                             │
│  1. 自动执行：条件满足时自动运行，无需人工干预               │
│  2. 不可篡改：代码一旦部署，无法修改                       │
│  3. 去中心化：运行在分布式网络上，无单点故障               │
│  4. 透明可信：代码开源，人人可验证                        │
│  5. 确定性执行：相同输入必产生相同输出                     │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

## 四、以太坊智能合约开发

### 4.1 Solidity 语言基础

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

// 简单代币合约
contract SimpleToken {
    string public name;
    string public symbol;
    uint8 public decimals;
    uint256 public totalSupply;
    
    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;
    
    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);
    
    constructor(uint256 _initialSupply) {
        name = "Simple Token";
        symbol = "STK";
        decimals = 18;
        totalSupply = _initialSupply * 10 ** uint256(decimals);
        balanceOf[msg.sender] = totalSupply;
    }
    
    function transfer(address to, uint256 amount) public returns (bool) {
        require(balanceOf[msg.sender] >= amount, "Insufficient balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        emit Transfer(msg.sender, to, amount);
        return true;
    }
    
    function approve(address spender, uint256 amount) public returns (bool) {
        allowance[msg.sender][spender] = amount;
        emit Approval(msg.sender, spender, amount);
        return true;
    }
    
    function transferFrom(address from, address to, uint256 amount) public returns (bool) {
        require(balanceOf[from] >= amount, "Insufficient balance");
        require(allowance[from][msg.sender] >= amount, "Not approved");
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        allowance[from][msg.sender] -= amount;
        emit Transfer(from, to, amount);
        return true;
    }
}
```

```solidity
contract Crowdfunding {
    address public beneficiary;
    uint256 public goal;
    uint256 public deadline;
    uint256 public raisedAmount;
    mapping(address => uint256) public contributions;
    
    event Contribution(address indexed contributor, uint256 amount);
    event Refund(address indexed contributor, uint256 amount);
    event Withdraw(address indexed beneficiary, uint256 amount);
    
    constructor(address _beneficiary, uint256 _goal, uint256 _duration) {
        beneficiary = _beneficiary;
        goal = _goal * 1 ether;
        deadline = block.timestamp + _duration;
    }
    
    function contribute() public payable {
        require(block.timestamp < deadline, "Crowdfunding ended");
        require(msg.value >= 0.1 ether, "Minimum contribution is 0.1 ETH");
        
        contributions[msg.sender] += msg.value;
        raisedAmount += msg.value;
        emit Contribution(msg.sender, msg.value);
    }
    
    function withdraw() public {
        require(msg.sender == beneficiary, "Not the beneficiary");
        require(raisedAmount >= goal, "Goal not reached");
        require(block.timestamp >= deadline, "Crowdfunding not ended");
        
        uint256 amount = raisedAmount;
        raisedAmount = 0;
        payable(beneficiary).transfer(amount);
        emit Withdraw(beneficiary, amount);
    }
    
    function refund() public {
        require(block.timestamp >= deadline && raisedAmount < goal, "Cannot refund");
        uint256 contributed = contributions[msg.sender];
        require(contributed > 0, "No contribution");
        
        contributions[msg.sender] = 0;
        payable(msg.sender).transfer(contributed);
        emit Refund(msg.sender, contributed);
    }
}
```

```typescript
// TypeScript/Web3.js 调用合约
import Web3 from 'web3';

const web3 = new Web3('https://mainnet.infura.io/v3/YOUR_PROJECT_ID');

// 合约 ABI
const abi = [
    {
        "inputs": [],
        "name": "name",
        "outputs": [{"type": "string"}],
        "stateMutability": "view",
        "type": "function"
    },
    {
        "inputs": [
            {"name": "to", "type": "address"},
            {"name": "amount", "type": "uint256"}
        ],
        "name": "transfer",
        "outputs": [{"type": "bool"}],
        "stateMutability": "nonpayable",
        "type": "function"
    }
];

const contractAddress = '0x1234567890123456789012345678901234567890';
const contract = new web3.eth.Contract(abi, contractAddress);

// 读取合约数据（view 函数）
async function getTokenInfo() {
    const name = await contract.methods.name().call();
    const symbol = await contract.methods.symbol().call();
    const totalSupply = await contract.methods.totalSupply().call();
    console.log(`Token: ${name} (${symbol}), Total Supply: ${totalSupply}`);
}

// 调用合约方法（state-changing 函数）
async function transfer(to: string, amount: number) {
    const accounts = await web3.eth.getAccounts();
    const from = accounts[0];
    
    const tx = contract.methods.transfer(to, amount);
    const gas = await tx.estimateGas({ from });
    const gasPrice = await web3.eth.getGasPrice();
    
    const result = await tx.send({
        from,
        gas,
        gasPrice
    });
    
    console.log('Transaction Hash:', result.transactionHash);
}
```

## 五、智能合约安全

### 5.1 常见漏洞

```solidity
// 重入攻击漏洞
contract VulnerableBank {
    mapping(address => uint256) public balances;
    
    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }
    
    // 漏洞：先转账，后更新状态
    function withdraw() public {
        uint256 amount = balances[msg.sender];
        require(amount > 0, "No balance");
        
        // 先转账，如果对方是合约，会调用 fallback 函数
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "Transfer failed");
        
        // 后更新状态，攻击者可在回调中再次调用
        balances[msg.sender] = 0;
    }
}

// 安全版本：使用 Checks-Effects-Interactions 模式
contract SecureBank {
    mapping(address => uint256) public balances;
    
    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }
    
    function withdraw() public {
        uint256 amount = balances[msg.sender];
        require(amount > 0, "No balance");
        
        // 先更新状态
        balances[msg.sender] = 0;
        
        // 后转账
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "Transfer failed");
    }
}
```

```solidity
// NFT 合约示例（ERC-721）
contract NFTContract is ERC721 {
    uint256 public tokenCounter;
    mapping(uint256 => string) private _tokenURIs;
    
    constructor() ERC721("MyNFT", "MNFT") {
        tokenCounter = 0;
    }
    
    function createNFT(string memory _tokenURI) public returns (uint256) {
        uint256 tokenId = tokenCounter;
        _safeMint(msg.sender, tokenId);
        _setTokenURI(tokenId, _tokenURI);
        tokenCounter++;
        return tokenId;
    }
    
    function _setTokenURI(uint256 tokenId, string memory _tokenURI) internal {
        require(_exists(tokenId), "URI set of nonexistent token");
        _tokenURIs[tokenId] = _tokenURI;
    }
}
```

## 七、面试常见问题

### Q1: 智能合约和传统合约的区别？

传统合约依赖法律体系和法院执行，智能合约依赖代码自动执行；智能合约不可篡改、执行成本低、无需信任第三方，但智能合约本身无法获取链外数据，需要预言机。

### Q2: 智能合约为什么需要 Gas？

Gas 是以太坊网络防止资源滥用的机制，每笔交易和合约调用都需要消耗 Gas 来支付计算和存储成本，防止无限循环和恶意攻击。

### Q3: 什么是预言机？

预言机（Oracle）是连接区块链和现实世界的桥梁，将链外数据（如价格、天气、体育比赛结果）引入链上，供智能合约使用。常见方案有 Chainlink、Band Protocol。

### Q4: 智能合约如何升级？

1. **代理模式**：逻辑合约 + 代理合约，升级逻辑合约地址
2. **钻石模式**：多个逻辑合约共享同一个代理
3. **版本控制**：部署新合约，迁移数据

### Q5: Solidity 的 storage 和 memory 区别？

- `storage`：永久存储在区块链上，消耗 Gas 高
- `memory`：临时存储在内存中，函数调用结束后释放
- `calldata`：函数参数专用，只读不复制，最省 Gas

## 八、总结

1. **智能合约定义**：部署在区块链上的可执行程序，条件触发自动执行
2. **核心特性**：自动执行、不可篡改、去中心化、透明可信
3. **开发语言**：Solidity（以太坊）、Rust（Solana）、Go（Fabric）
4. **安全要点**：防止重入、整数溢出、权限控制、输入验证
5. **应用场景**：DeFi、NFT、DAO、游戏、供应链

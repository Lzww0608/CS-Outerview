# EIP-4337 账户抽象与 UserOperation 执行流程

## 1. 问题背景

EIP-4337 是以太坊账户抽象的核心提案，允许用户使用智能合约钱包而非传统 EOA（Externally Owned Account）进行交易签署，实现social recovery、multisig、gas代付等高级功能，无需修改以太坊核心协议。

```text
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  传统以太坊账户 vs 合约钱包：                              │
│                                                             │
│  传统 EOA:                                                │
│  • 私钥控制                                              │
│  • 每笔交易需私钥签名                                    │
│  • 无法自定义验证逻辑                                    │
│  • 无法支付他人 gas                                      │
│                                                             │
│  EIP-4337 合约钱包:                                       │
│  • 合约控制                                              │
│  • 验证逻辑可编程                                        │
│  • 支持多签/社交恢复                                    │
│  • 可由第三方代付 gas                                    │
│  • 无需修改共识层                                        │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

---

## 2. 问题现象

| 现象 | 传统 EOA | EIP-4337 合约钱包 |
|------|----------|-------------------|
| 交易签署 | 必须私钥签名 | 可自定义验证逻辑 |
| Gas 支付 | 必须有 ETH | 可由他人代付 |
| 多签支持 | 需要多签合约 | 原生支持 |
| 密钥恢复 | 无 | 社交恢复 |
| 批量交易 | 需多个签名 | 一次签名多个操作 |

---

## 3. 影响范围

- 钱包应用（MetaMask, Rainbow 等）
- DeFi 用户体验优化
- 机构资产管理
- 游戏内道具购买
- 订阅服务自动扣款

### 4.3 UserOperation 结构

```solidity
// UserOperation 结构定义

struct UserOperation {
    address sender;              // 钱包合约地址
    uint256 nonce;               // 防重放 nonce
    bytes initCode;              // 钱包部署代码（首次创建时）
    bytes callData;              // 要执行的数据
    uint256 callGasLimit;        // 调用 gas 上限
    uint256 verificationGasLimit;// 验证 gas 上限
    uint256 preVerificationGas;  // 预验证 gas
    uint256 maxFeePerGas;        // 最大 gas 费用
    uint256 maxPriorityFeePerGas;// 最大优先费
    bytes paymasterAndData;       // Paymaster 相关数据
    bytes signature;             // 用户签名
}
```

## 5. 解决方案

### 5.1 SimpleAccount 合约实现

```solidity
// 简单账户合约示例

contract SimpleAccount is IAccount {
    address public owner;
    address public entryPoint;
    uint256 public nonce;
    
    modifier onlyEntryPoint() {
        require(msg.sender == entryPoint, "only entryPoint");
        _;
    }
    
    constructor(address _owner, address _entryPoint) {
        owner = _owner;
        entryPoint = _entryPoint;
    }
    
    // EIP-4337 核心：验证 UserOperation
    function validateUserOp(
        UserOperation calldata op,
        bytes32 opHash,
        uint256 missingAccountFunds
    ) external onlyEntryPoint returns (uint256) {
        // 1. 验证签名
        bytes32 hash = keccak256(abi.encode(opHash, address(this), nonce));
        
        if (recoverSigner(hash, op.signature) != owner) {
            return 1;  // 签名失败
        }
        
        // 2. 验证 nonce（简化版）
        // 实际实现需要更复杂的 nonce 管理
        
        // 3. 支付 Gas（可选）
        if (missingAccountFunds > 0) {
            payable(msg.sender).call{value: missingAccountFunds}("");
        }
        
        nonce++;
        return 0;  // 验证成功
    }
    
    // 执行用户操作
    function execute(bytes32 opHash, bytes calldata data) external onlyEntryPoint {
        (address target, uint256 value, bytes memory callData) = 
            abi.decode(data, (address, uint256, bytes));
        
        (bool success, ) = target.call{value: value}(callData);
        require(success, "call failed");
    }
    
    // 签名恢复
    function recoverSigner(bytes32 hash, bytes memory signature) 
        internal 
        pure 
        returns (address) 
    {
        bytes32 r;
        bytes32 s;
        uint8 v;
        
        (r, s, v) = splitSignature(signature);
        return ecrecover(hash, v, r, s);
    }
    
    function splitSignature(bytes memory sig) 
        internal 
        pure 
        returns (bytes32 r, bytes32 s, uint8 v) 
    {
        require(sig.length == 65, "invalid signature length");
        
        assembly {
            r := mload(add(sig, 32))
            s := mload(add(sig, 64))
            v := byte(0, mload(add(sig, 96)))
        }
    }
}
```

### 5.2 Paymaster 实现（Gas 代付）

```solidity
// Paymaster 示例：允许 DApp 代付 Gas

contract VerifyingPaymaster is IPaymaster {
    address public entryPoint;
    address public verifyingSigner;
    uint256 public stake;
    
    mapping(address => uint256) public prepaid;
    
    constructor(address _entryPoint, address _verifyingSigner) {
        entryPoint = _entryPoint;
        verifyingSigner = _verifyingSigner;
    }
    
    // 验证 UserOperation（是否允许代付）
    function validatePaymasterUserOp(
        UserOperation calldata op,
        bytes32 opHash,
        uint256 maxCost
    ) external returns (bytes memory context, uint256 validationData) {
        // 1. 验证签名（确保请求来自授权应用）
        bytes32 hash = keccak256(abi.encode(opHash, address(this)));
        require(
            recoverSigner(hash, op.paymasterAndData[64:]) == verifyingSigner,
            "invalid signature"
        );
        
        // 2. 检查是否有足够余额
        uint256 sponsor = abi.decode(op.paymasterAndData[0:32], (uint256));
        require(prepaid[sponsor] >= maxCost, "insufficient balance");
        
        // 3. 扣除预付金额
        prepaid[sponsor] -= maxCost;
        
        return (abi.encode(sponsor), 0);
    }
    
    // 执行后处理
    function postOp(
        PostOpMode mode,
        bytes calldata context,
        uint256 actualGasCost
    ) external {
        address sponsor = abi.decode(context, (address));
        
        // 根据实际消耗结算
        if (mode == PostOpMode.postOpReverted) {
            // 操作失败，退款
        } else {
            // 操作成功，记录日志
        }
    }
}
```

### 5.3 Bundler 集成

```typescript
// Bundler 客户端示例

import { ethers } from 'ethers';

class EIP4337Bundler {
    private entryPoint: ethers.Contract;
    private provider: ethers.JsonRpcProvider;
    
    constructor(
        rpcUrl: string,
        entryPointAddress: string
    ) {
        this.provider = new ethers.JsonRpcProvider(rpcUrl);
        this.entryPoint = new ethers.Contract(
            entryPointAddress,
            ENTRYPOINT_ABI,
            this.provider
        );
    }
    
    // 发送 UserOperation
    async sendUserOperation(
        userOp: UserOperation,
        bundlerKey: string
    ): Promise<string> {
        const wallet = new ethers.Wallet(bundlerKey, this.provider);
        
        // 估算 Gas
        const gasEstimate = await this.provider.send(
            'eth_estimateUserOperationGas',
            [userOp, this.entryPoint.target]
        );
        
        userOp.preVerificationGas = gasEstimate.preVerificationGas;
        userOp.verificationGasLimit = gasEstimate.verificationGas;
        userOp.callGasLimit = gasEstimate.callGasLimit;
        
        // 获取 Gas 价格
        const feeData = await this.provider.getFeeData();
        userOp.maxFeePerGas = feeData.maxFeePerGas!;
        userOp.maxPriorityFeePerGas = feeData.maxPriorityFeePerGas!;
        
        // 发送
        const tx = await this.entryPoint
            .connect(wallet)
            .handleOps([userOp], wallet.address);
        
        return tx.hash;
    }
    
    // 模拟验证
    async simulateValidation(userOp: UserOperation): Promise<SimulationResult> {
        return await this.provider.send(
            'eth_simulateValidation',
            [userOp]
        );
    }
}
```

## 11. 总结

1. **EIP-4337 核心**：无需改共识层实现账户抽象
2. **UserOperation**：类比交易但更灵活的结构
3. **执行流程**：Bundler 收集 → EntryPoint 验证 → 执行 → 结算
4. **核心合约**：EntryPoint、Wallet、Paymaster
5. **应用场景**：Gas 代付、社交恢复、多签、批量交易

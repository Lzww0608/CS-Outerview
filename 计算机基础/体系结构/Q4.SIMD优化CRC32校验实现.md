# SIMD 优化 CRC32 校验实现

## 一、CRC32 基础原理

### 1.1 CRC32 定义

CRC32（循环冗余校验）是一种广泛用于数据校验的算法，常用于网络协议、压缩文件、存储设备等场景。

```c
// 标准 CRC32 多项式：0xEDB88320（reflected）
// 或：0x04C11DB7（非反射）
```

### 1.2 传统查表法

```c
// 传统 CRC32：每次处理 1 字节
static uint32_t crc32_table[256];

void init_crc32_table() {
    for (int i = 0; i < 256; i++) {
        uint32_t crc = i;
        for (int j = 0; j < 8; j++) {
            if (crc & 1) crc = (crc >> 1) ^ 0xEDB88320;
            else crc >>= 1;
        }
        crc32_table[i] = crc;
    }
}

uint32_t crc32_slow(const uint8_t* data, size_t len) {
    uint32_t crc = 0xFFFFFFFF;
    for (size_t i = 0; i < len; i++) {
        crc = crc32_table[(crc ^ data[i]) & 0xFF] ^ (crc >> 8);
    }
    return ~crc;
}
```

## 二、SIMD 优化原理

### 2.1 使用 PCLMULQDQ 指令

Intel 的 `PCLMULQDQ`（Packed Carry-Less Multiplication Quadword）指令可以并行计算多个 64 位字段的无进位乘法，非常适合 CRC 计算。

```text
PCLMULQDQ 原理：
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  XMM 寄存器（128位）                                        │
│  ┌──────────────────┬──────────────────┐                  │
│  │    64-bit        │    64-bit        │                  │
│  └──────────────────┴──────────────────┘                  │
│                                                             │
│  可以同时计算两个 64 位的 GF(2) 多项式乘法                   │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 2.2 CRC32 指令

Intel SSE4.2 引入了 `CRC32` 指令（`CRC32B`、`CRC32D`、`CRC32Q`），可以在一个指令周期内完成 8/32/64 位数据的 CRC 计算。

```c
// GCC/Clang Intrinsics
#include <nmmintrin.h>

uint32_t crc32_1byte(uint32_t crc, uint8_t data) {
    return _mm_crc32_u8(crc, data);
}

uint32_t crc32_4bytes(uint32_t crc, uint32_t data) {
    return _mm_crc32_u32(crc, data);
}

uint32_t crc32_8bytes(uint32_t crc, uint64_t data) {
    return _mm_crc32_u64(crc, data);
}
```

## 三、C++ 实现

### 3.1 单指令 CRC32（SSE4.2）

```c++
#include <nmmintrin.h>
#include <cstdint>
#include <cstddef>

class CRC32SSE {
public:
    // 8字节并行 CRC32
    static uint32_t crc32_avx(const uint8_t* data, size_t len, uint32_t init = 0xFFFFFFFF) {
        uint32_t crc = init;
        size_t i = 0;
        
        // 对齐到 8 字节
        for (; i + 8 <= len && (reinterpret_cast<size_t>(data + i) & 7); i++) {
            crc = _mm_crc32_u8(crc, data[i]);
        }
        
        // 8 字节批量处理
        for (; i + 8 <= len; i += 8) {
            uint64_t val = *reinterpret_cast<const uint64_t*>(data + i);
            crc = _mm_crc32_u64(crc, val);
        }
        
        // 处理剩余字节
        for (; i < len; i++) {
            crc = _mm_crc32_u8(crc, data[i]);
        }
        
        return ~crc;
    }
    
    // 16 字节并行（SSE 16字节 + PCLMULQDQ）
    static uint32_t crc32_pclmul(const uint8_t* data, size_t len, uint32_t init = 0xFFFFFFFF) {
        uint32_t crc = init;
        size_t i = 0;
        
        // 16 字节对齐
        for (; i + 16 <= len && (reinterpret_cast<size_t>(data + i) & 15); i++) {
            crc = _mm_crc32_u8(crc, data[i]);
        }
        
        // 16 字节批量处理
        for (; i + 16 <= len; i += 16) {
            __m128i block = _mm_loadu_si128(reinterpret_cast<const __m128i*>(data + i));
            uint64_t lo = _mm_extract_epi64(block, 0);
            uint64_t hi = _mm_extract_epi64(block, 1);
            crc = _mm_crc32_u64(crc, lo);
            crc = _mm_crc32_u64(crc, hi);
        }
        
        // 处理剩余字节
        for (; i < len; i++) {
            crc = _mm_crc32_u8(crc, data[i]);
        }
        
        return ~crc;
    }
};
```

### 3.2 PCLMULQDQ 完整优化

```c++
#include <wmmintrin.h>
#include <immintrin.h>
#include <smmintrin.h>
#include <cstdint>
#include <cstddef>
#include <cstring>

class CRC32PCLMUL {
private:
    // GF(2) 多项式乘法
    static __m128i gf2_poly_mul(__m128i a, __m128i b) {
        return _mm_clmulepi64_si128(a, b, 0x00);
    }
    
    // Barrett 约减
    static __m128i barrett_reduction(__m128i x, __m128i mod, __m128i mu) {
        __m128i t1 = _mm_clmulepi64_si128(x, mu, 0x10);
        t1 = _mm_clmulepi64_si128(t1, mod, 0x00);
        t1 = _mm_srli_si128(t1, 8);
        x = _mm_xor_si128(x, t1);
        t1 = _mm_clmulepi64_si128(t1, mod, 0x00);
        t1 = _mm_srli_si128(t1, 8);
        return _mm_xor_si128(x, t1);
    }
    
public:
    // 使用 PCLMULQDQ 的 CRC32（每次处理 16 字节）
    static uint32_t crc32_slices(const uint8_t* data, size_t len, uint32_t init = 0xFFFFFFFF) {
        // 常量
        const __m128i K1 = _mm_set_epi32(0, 0, 0x1C6E415D, 0x1C6E415D); // P(x)
        const __m128i K2 = _mm_set_epi32(0x1C6E415D, 0, 0, 0);         // 约减参数
        const __m128i M1 = _mm_set_epi32(0x1C6E415D, 0xF7011641, 0, 0);
        
        __m128i r = _mm_set_epi32(init, init, init, init);
        size_t i = 0;
        
        // 对齐到 16 字节
        for (; i + 16 <= len && (reinterpret_cast<size_t>(data + i) & 15); i++) {
            r = _mm_xor_si128(r, _mm_cvtsi32_si128(_mm_crc32_u32(0, data[i])));
        }
        
        // 16 字节批量处理（使用 PCLMULQDQ）
        for (; i + 16 <= len; i += 16) {
            __m128i x = _mm_loadu_si128(reinterpret_cast<const __m128i*>(data + i));
            
            // 分裂转换
            __m128i a1 = _mm_unpacklo_epi8(x, _mm_setzero_si128());
            __m128i a2 = _mm_unpackhi_epi8(x, _mm_setzero_si128());
            a1 = _mm_unpacklo_epi16(a1, _mm_setzero_si128());
            a2 = _mm_unpackhi_epi16(a1, _mm_setzero_si128());
            
            // GF(2) 乘法
            __m128i t = gf2_poly_mul(_mm_cvtsi32_si128(init), K1);
            t = _mm_xor_si128(t, x);
            t = gf2_poly_mul(t, K2);
            t = barrett_reduction(t, K1, M1);
            
            r = _mm_xor_si128(r, t);
        }
        
        // 处理剩余字节
        uint32_t crc = _mm_extract_epi32(r, 0);
        for (; i < len; i++) {
            crc = _mm_crc32_u8(crc, data[i]);
        }
        
        return ~crc;
    }
};
```

## 四、Golang 实现

### 4.1 纯 Go 查表法

```go
package crc32

// CRC32 多项式
const (
    IEEE = 0xEDB88320
    Castagnoli = 0x82F63B78
)

var table [256]uint32

func init() {
    for i := range table {
        crc := uint32(i)
        for j := 0; j < 8; j++ {
            if crc&1 == 1 {
                crc = (crc >> 1) ^ IEEE
            } else {
                crc >>= 1
            }
        }
        table[i] = crc
    }
}

// 查表法
func CRC32(data []byte) uint32 {
    crc := ^uint32(0)
    for _, b := range data {
        crc = table[(crc^uint32(b))&0xFF] ^ (crc >> 8)
    }
    return ^crc
}
```

### 4.2 使用 Go asm 汇编优化

```go
// crc32_amd64.go
package crc32

//go:noescape
func archChecksumAVX(data []byte, init uint32) uint32

// crc32_amd64.s (手写汇编)
TEXT ·archChecksumAVX(SB), NOSPLIT, $0-40
    MOVQ    data+0(FP), DI      // DI = data pointer
    MOVQ    data+8(FP), SI       // SI = len
    MOVL    init+16(FP), AX     // AX = init CRC
    
    // 使用 CRC32 指令
    XORL    AX, AX              // 确保 AX 初始化
    MOVQ    SI, BX              // BX = remaining length
    
    // 对齐到 8 字节
    TESTQ   $7, DI
    JZ      aligned
unaligned:
    CRC32B  (DI), AX
    INCQ    DI
    DECQ    BX
    TESTQ   $7, DI
    JNZ     unaligned
    
aligned:
    // 8 字节批量处理
    CMPQ    BX, $8
    JB      tail
loop8:
    CRC32Q  (DI), AX
    ADDQ    $8, DI
    SUBQ    $8, BX
    CMPQ    BX, $8
    JAE     loop8
    
tail:
    // 处理剩余字节
    CMPQ    BX, $0
    JZ      done
tail_loop:
    CRC32B  (DI), AX
    INCQ    DI
    DECQ    BX
    JNZ     tail_loop
    
done:
    NOTL    AX
    MOVL    AX, ret+24(FP)
    RET
```

```go
    // 使用 cpuid 指令检测

    return haveSSE42

//go:linkname haveSSE42 runtime.haveSSE42
func haveSSE42() bool

func Checksum(data []byte, init uint32) uint32 {
    if HaveSSE42() {
        return archChecksumAVX(data, init)
```

### 4.4 使用 golang.org/x/crypto/sha256 的 CRC32

```go
package main

import (
    "fmt"
    "hash/crc32"
)

func main() {
    data := []byte("Hello, World!")
    
    // 使用标准库（已优化）
    ieee := crc32.NewIEEE()
    ieee.Write(data)
    fmt.Printf("CRC32-IEEE: 0x%08X\n", ieee.Sum32())
    
    // Castagnoli 多项式（更常用，纠错能力更强）

    fmt.Printf("CRC32-Castagnoli: 0x%08X\n", cast.Sum32())
}
```

## 六、面试常见问题

### Q1: CRC32 的数学原理是什么？

CRC 是基于有限域 GF(2) 的多项式除法。将数据视为多项式，除以生成多项式，余数就是 CRC 值。使用模 2 算术（加法=异或，减法=异或）。

### Q2: SIMD 如何加速 CRC32？

使用 Intel SSE4.2 的 `CRC32` 指令，可以一个指令周期计算 8/32/64 位数据的 CRC。对于更长数据，可以使用 `PCLMULQDQ` 指令并行计算多个多项式乘法。

### Q3: CRC32 和 CRC16 的区别？

| 特性 | CRC16 | CRC32 |
|------|-------|-------|
| 位宽 | 16 位 | 32 位 |
| 多项式 | 常用 CRC16-CCITT | IEEE 802.3 (0x04C11DB7) |
| 检错能力 | 99.99% | 99.9999% |
| 性能 | 更快 | 略慢 |
| 应用 | Modbus、USB | Ethernet、PNG、ZIP |

### Q4: 如何选择 CRC 多项式？

1. **IEEE 802.3**：以太网、ZIP、PNG（通用）
2. **Castagnoli**：高效硬件实现（iSCSI、文件存储）
3. **Koopman**：最优 Hamming 距离（需要最高可靠性）
